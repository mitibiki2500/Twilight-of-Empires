	visible = { this = c:PAI NOT = { has_variable = toe_paido_done_dale } NOT = { has_variable = toe_paido_finalize_integration } }
	possible = {
		custom_tooltip = { text = toe_paido_dale_agri_tt var:toe_pai_dale_agri >= 50 }
		custom_tooltip = { text = toe_paido_dale_food_tt var:toe_pai_dale_food >= 10 }
	}
	effect = { toe_paido_integrate_dale = yes }
	ai_chance = { base = 100 }
}

# 必須6文化を統合した後も任意文化を待てるよう、最後は手動で完了する。
REPLACE_OR_CREATE:toe_paido_finish_integration_button = {
	visible = {
		this = c:PAI
		NOT = { has_variable = toe_paido_finalize_integration }
	}
	possible = {
		custom_tooltip = { text = toe_paido_req_dale_done_tt has_variable = toe_paido_done_dale }
		custom_tooltip = { text = toe_paido_req_shaftar_done_tt has_variable = toe_paido_done_shaftar }
		custom_tooltip = { text = toe_paido_req_eskan_done_tt has_variable = toe_paido_done_eskan }
		custom_tooltip = { text = toe_paido_req_old_done_tt has_variable = toe_paido_done_old }
		custom_tooltip = { text = toe_paido_req_northern_done_tt has_variable = toe_paido_done_northern }
		custom_tooltip = { text = toe_paido_req_palun_done_tt has_variable = toe_paido_done_palun }
		custom_tooltip = { text = toe_paido_req_nationalism_tt has_technology_researched = nationalism }
		custom_tooltip = { text = toe_paido_req_fervor_tt cu:nsm_map_pai.culture_current_fervor >= 60 }
		custom_tooltip = {
			text = toe_paido_req_schools_tt
			institution_investment_level = { institution = institution_schools value >= 2 }
		}
	}
	effect = { set_variable = toe_paido_finalize_integration }
	ai_chance = { base = 100 }
}
