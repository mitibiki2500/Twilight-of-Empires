# Alti Great Conquest claim buttons.
# Players choose when to pay 20 infamy and receive claims. AI is hard-blocked at 70+ infamy.

REPLACE_OR_CREATE:toe_alti_claim_sparia_button = {
    visible = {
        this = c:SKR
        NOT = { has_variable = toe_altia_sparia_claims_granted }
    }

    possible = {
        OR = {
            is_ai = no
            infamy < 70
        }
    }

    effect = {
        toe_alti_grant_sparia_claims = yes
    }

    ai_chance = {
        base = 100
        modifier = {
            factor = 0
            infamy >= 70
        }
    }
}

REPLACE_OR_CREATE:toe_alti_claim_gorton_button = {
    visible = {
        this = c:SKR
        NOT = { has_variable = toe_altia_gorton_claims_granted }
    }

    possible = {
        # 大ジュズを傀儡化済み、または大ジュズの2州を直接支配していること。
        OR = {
            has_diplomatic_pact = { who = c:XGG type = puppet }
            AND = {
                owns_entire_state_region = STATE_NSM_258
                owns_entire_state_region = STATE_NSM_430
            }
        }
        # 1850年までの初期停戦中は押せない。
        NOT = { has_truce_with = c:GOT }
        OR = {
            is_ai = no
            infamy < 70
        }
    }

    effect = {
        # 既存の請求権・悪名処理は維持する。
        toe_alti_grant_gorton_claims = yes
        # ゴートン併合の外交戦を開始する。
        c:GOT = {
            create_diplomatic_play = {
                name = toe_alti_gorton_annexation_play
                initiator = c:SKR
                target_country = c:GOT
                type = dp_annex_war
                war = no
            }
        }
    }

    ai_chance = {
        base = 100
        modifier = { factor = 0 infamy >= 70 }
    }
}
