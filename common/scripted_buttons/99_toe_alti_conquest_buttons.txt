# Alti Great Conquest claim buttons.
# Each button costs 20 infamy. AI may press only at 80 infamy or below, so the result never exceeds 100.

REPLACE_OR_CREATE:toe_alti_claim_sparia_button = {
    visible = { this = c:SKR NOT = { has_variable = toe_altia_sparia_claims_granted } }
    possible = { OR = { is_ai = no infamy <= 80 } }
    effect = { toe_alti_grant_sparia_claims = yes }
    ai_chance = { base = 1000 modifier = { factor = 0 infamy > 80 } }
}

REPLACE_OR_CREATE:toe_alti_claim_gorton_button = {
    visible = { this = c:SKR NOT = { has_variable = toe_altia_gorton_claims_granted } }
    possible = {
        # 大ジュズを傀儡化済み、または大ジュズの2州を直接支配していること。
        OR = {
            has_diplomatic_pact = { who = c:XGG type = puppet }
            AND = {
                owns_entire_state_region = STATE_NSM_258
                owns_entire_state_region = STATE_NSM_430
            }
        }
        # ゴートン改革JEの失敗後にのみ一発併合を解禁する。
        c:GOT ?= { has_variable = toe_gorton_reform_failed }
        # 1850年までの初期停戦中は押せない。
        NOT = { has_truce_with = c:GOT }
        OR = { is_ai = no infamy <= 80 }
    }
    effect = {
        toe_alti_grant_gorton_claims = yes
        c:GOT = {
            create_diplomatic_play = {
                name = toe_alti_gorton_annexation_play
                initiator = c:SKR
                target_country = c:GOT
                type = dp_annex_war
                war = no
            }
        }
    }
    ai_chance = { base = 1000 modifier = { factor = 0 infamy > 80 } }
}
