# ToE: Gorton reform difficulty / Alti annexation gate compatibility
# The player can still complete the 25-year reform journal normally.
# AI Gorton cannot complete the main reform goal, so it will inevitably time out.

REPLACE_OR_CREATE:je_sick_man_main = {
    name = je_sick_man_main_name
    desc = je_sick_man_main_reason
    icon = "gfx/interface/icons/event_icons/event_skull.dds"

    group = je_group_tanzimat

    is_shown_in_lobby = {
        c:GOT = THIS
    }

    complete = {
        # Player-only completion. AI Gorton is intentionally forced to fail this JE.
        is_ai = no
        scope:journal_entry = {
            is_goal_complete = yes
        }
    }

    on_complete = {
        remove_modifier = toe_gorton_sick_man_of_europe
        trigger_event = {
            id = toe_gorton_tanzimat.2
        }
    }

    # 25 years.
    timeout = 9125
    on_timeout = {
        remove_modifier = toe_gorton_sick_man_of_europe
        add_modifier = {
            name = toe_gorton_reform_failure
            days = 7300
            is_decaying = yes
        }
        # Existing completion marker remains for the sub-JEs.
        set_variable = sick_man_complete
        # Dedicated gate used by Alti's one-shot Gorton annexation route.
        set_variable = toe_gorton_reform_failed
        trigger_event = {
            id = toe_gorton_tanzimat.3
        }
    }

    current_value = {
        value = var:sick_man_var
    }

    # Five reform objectives are still required for a human player.
    goal_add_value = {
        add = 5
    }

    progressbar = yes
    weight = 10000
    should_be_pinned_by_default_uninvolved_or_context = yes
}
