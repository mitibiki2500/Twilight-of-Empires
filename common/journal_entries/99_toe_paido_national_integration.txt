# Paido-specific modernization journals.

je_toe_paido_cultural_integration = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_historical_content

	is_shown_in_lobby = { c:PAI ?= this }

	immediate = {
		toe_paido_init_integration = yes
	}

	# 各文化の母国州にある施設レベルを月1回再集計する。
	# 全保有州→全建物を走査するため、週次から月次へ変更して負荷を削減。
	on_monthly_pulse = {
		effect = {
			if = {
				limit = { NOT = { has_variable = toe_paido_finalize_integration } }
				toe_paido_recalculate_integration = yes
			}
		}
	}

	scripted_button = toe_paido_integrate_dale_button
	scripted_button = toe_paido_integrate_shaftar_button
	scripted_button = toe_paido_integrate_eskan_button
	scripted_button = toe_paido_integrate_old_button
	scripted_button = toe_paido_integrate_northern_button
	scripted_button = toe_paido_integrate_palun_button
	scripted_button = toe_paido_tradition_resind_button
	scripted_button = toe_paido_tradition_prune_button
	scripted_button = toe_paido_tradition_krikul_button
	scripted_button = toe_paido_tradition_sosilia_button
	scripted_button = toe_paido_integrate_balin_button
	scripted_button = toe_paido_finish_integration_button

	complete = {
		custom_tooltip = {
			text = toe_paido_integration_finalized_tt
			has_variable = toe_paido_finalize_integration
		}
	}

	on_complete = {
		toe_paido_apply_integration_reward = yes
	}

	timeout = 9125
	on_timeout = {
		add_modifier = {
			name = toe_paido_integration_failure
			days = 3650
		}
	}

	invalid = { NOT = { c:PAI ?= this } }
	weight = 100
	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_toe_paido_monarchist_rivalry = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"
	group = je_group_historical_content

	is_shown_in_lobby = { c:PAI ?= this }

	modifiers_while_active = { toe_paido_rival_monarchs }

	# ToEでは開始時にleader未生成のIGが存在するため、
	# 史実フランス用 fra_* scripted effect は使用しない。
	# これらは全IGのleader存在を前提にしており、unset scopeを大量発生させていた。

	scripted_progress_bar = divided_monarchists_orleanist_progress_bar
	scripted_progress_bar = divided_monarchists_legitimist_progress_bar

	on_monthly_pulse = {
		effect = {
			set_variable = { name = toe_paido_military_facility_levels value = 0 }
			every_scope_building = {
				limit = { is_building_group = bg_military }
				ROOT = { change_variable = { name = toe_paido_military_facility_levels add = PREV.level } }
			}
			scope:journal_entry = {
				add_progress = {
					name = divided_monarchists_orleanist_progress_bar
					value = root.var:toe_paido_military_facility_levels
				}
			}
		}
	}

	complete = {
		country_has_monarchy_law = yes
		scope:journal_entry = {
			OR = {
				"scripted_bar_progress(divided_monarchists_orleanist_progress_bar)" >= divided_monarchists_upper_bound
				"scripted_bar_progress(divided_monarchists_legitimist_progress_bar)" >= divided_monarchists_upper_bound
			}
		}
	}

	on_complete = {
		if = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(divided_monarchists_orleanist_progress_bar)" > "scripted_bar_progress(divided_monarchists_legitimist_progress_bar)"
				}
			}
			set_variable = toe_paido_innovative_monarchs_won
		}
		else = { set_variable = toe_paido_conservative_monarchs_won }
	}

	invalid = { NOT = { country_has_monarchy_law = yes } }

	weight = 100
	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}
