    }

    on_complete = {
        set_variable = toe_altia_sparia_conquered
        toe_alti_reward_sparia = yes
        if = { limit = { NOT = { has_journal_entry = je_nsm_altia_conquest_altia } } add_journal_entry = { type = je_nsm_altia_conquest_altia } }
