# ToE: Alti AI conquest progression
# AI should not depend on scripted-button clicks to start the Great Conquest journals.

on_monthly_pulse = {
    on_actions = {
        toe_alti_ai_conquest_pulse
    }
}

toe_alti_ai_conquest_pulse = {
    effect = {
        c:SKR ?= {
            if = {
                limit = {
                    is_ai = yes
                    infamy < 70
                    NOT = { has_variable = toe_altia_sparia_claims_granted }
                }
                toe_alti_grant_sparia_claims = yes
            }

            if = {
                limit = {
                    is_ai = yes
                    infamy < 70
                    has_variable = toe_altia_sparia_conquered
                    NOT = { has_variable = toe_altia_altia_claims_granted }
                }
                toe_alti_grant_altia_claims = yes
            }

            if = {
                limit = {
                    is_ai = yes
                    infamy < 70
                    has_variable = toe_altia_sparia_conquered
                    NOT = { has_variable = toe_altia_gorton_claims_granted }
                    NOT = { has_truce_with = c:GOT }
                    is_at_war = no
                    c:GOT ?= {
                        exists = yes
                        has_variable = toe_gorton_reform_failed
                    }
                }
                toe_alti_grant_gorton_claims = yes
                c:GOT = {
                    create_diplomatic_play = {
                        name = toe_alti_gorton_annexation_play
                        initiator = c:SKR
                        target_country = c:GOT
                        type = dp_annex_war
                        war = no
                    }
                }
            }

            if = {
                limit = {
                    is_ai = yes
                    infamy < 70
                    has_variable = toe_altia_altia_conquered
                    has_variable = toe_altia_gorton_conquered
                    NOT = { has_variable = toe_altia_paido_claims_granted }
                }
                toe_alti_grant_paido_claims = yes
            }

            if = {
                limit = {
                    is_ai = yes
                    infamy < 70
                    has_variable = toe_altia_paido_conquered
                    NOT = { has_variable = toe_altia_trilia_claims_granted }
                }
                toe_alti_grant_trilia_claims = yes
            }

            if = {
                limit = {
                    is_ai = yes
                    infamy < 70
                    has_variable = toe_altia_paido_conquered
                    NOT = { has_variable = toe_altia_ilvone_claims_granted }
                }
                toe_alti_grant_ilvone_claims = yes
            }
        }
    }
}
