# Kamehameha coin toss
kamehameha_coin_toss = {
	trigger = {
		c:HAW ?= this
		has_law = law_type:law_monarchy
		ruler = {
			has_template = HAW_kamehameha_iii
		}
		year >= 1850
		year <= 1855
	}
	effect = {
		random = {
			chance = 50

			ruler = {
				kill_character = yes
			}
		}
	}
}

	trigger = {
		OR = {
			c:CHI ?= this
			c:JAP ?= this
		}
        AND = {
            c:JAP ?= { has_modifier = modifier_ryukyu_rivalry_officer_training }
            c:CHI ?= { has_modifier = modifier_ryukyu_rivalry_officer_training }
        }
		NOT = { has_global_variable = ryukyu_rivalry_training_incident }
	}
	effect = {
		set_global_variable = {
			name = ryukyu_rivalry_training_incident
			years = 5
		}

			1 = {
				set_global_variable = {
					name = ryukyu_rivalry_fire
				}

				}
			}
