# ToE 1.13 compatibility
# XAI (Tegaar) / XBJ (Ria Republic) can start with an empty IG leader.
# Vanilla party scripts assume a valid leader scope; fill only genuinely empty IGs.

on_game_started = {
    effect = {
        every_country = {
            limit = {
                OR = {
                    c:XAI ?= this
                    c:XBJ ?= this
                }
            }
            every_interest_group = {
                limit = {
                    NOT = {
                        leader ?= { always = yes }
                    }
                }
                save_scope_as = toe_missing_ig_leader_group
                owner = {
                    create_character = {
                        age = 40
                        interest_group = scope:toe_missing_ig_leader_group
                        ideology = ideology_moderate
                        ig_leader = yes
                    }
                }
            }
        }
    }
}
