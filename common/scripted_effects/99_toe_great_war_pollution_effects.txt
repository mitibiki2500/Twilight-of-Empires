# Scope: war. scope:state is supplied by on_battle_ended.
# Chemical-weapon battle contamination uses combined war casualties only as a severity proxy.
# Thresholds are intentionally conservative because the effect can fire after multiple gas battles.
toe_apply_gas_battle_pollution = {
    if = {
        limit = { num_casualties >= 2000000 }
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 10 duration = 9125 } } }
    }
    else_if = {
        limit = { num_casualties >= 1500000 }
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 9 duration = 7300 } } }
    }
    else_if = {
        limit = { num_casualties >= 1000000 }
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 8 duration = 6570 } } }
    }
    else_if = {
        limit = { num_casualties >= 750000 }
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 7 duration = 5840 } } }
    }
    else_if = {
        limit = { num_casualties >= 500000 }
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 6 duration = 5110 } } }
    }
    else_if = {
        limit = { num_casualties >= 350000 }
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 5 duration = 4380 } } }
    }
    else_if = {
        limit = { num_casualties >= 200000 }
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 4 duration = 3650 } } }
    }
    else_if = {
        limit = { num_casualties >= 100000 }
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 3 duration = 2920 } } }
    }
    else_if = {
        limit = { num_casualties >= 50000 }
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 2 duration = 2190 } } }
    }
    else = {
        scope:state = { state_region = { start_harvest_condition_with_params = { type = toe_great_war_compound_contamination intensity = 1 duration = 1825 } } }
    }
}
