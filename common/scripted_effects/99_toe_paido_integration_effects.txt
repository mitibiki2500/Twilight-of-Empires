# Paido cultural integration helpers.
# 「母国の施設Lv」は、PAIが保有するその文化のhomeland州すべての合計で数える。

toe_paido_init_integration = {
	set_variable = { name = toe_paido_achieved_count value = 0 }
	toe_paido_recalculate_integration = yes
}

# 文化統合（POPをパイドへ全転換＋tradition_paido付与）
toe_paido_integrate_dale = {
	cu:nsm_map_pai_dale = { add_tradition_trait = tradition_paido }
	every_scope_state = { every_scope_pop = { limit = { has_pop_culture = cu:nsm_map_pai_dale } change_pop_culture = { target = cu:nsm_map_pai value = 1 } } }
	set_variable = toe_paido_done_dale
	change_variable = { name = toe_paido_achieved_count add = 1 }
}

# 非必須：文化は残してtradition_paidoだけ付与。
toe_paido_tradition_resind = { cu:nsm_map_pai_3 = { add_tradition_trait = tradition_paido } set_variable = toe_paido_done_resind change_variable = { name = toe_paido_achieved_count add = 1 } }

# 毎週、各文化のhomeland内の施設レベルを再集計して条件を満たした文化を自動達成する。
toe_paido_recalculate_integration = {
	# 必須文化
	set_variable = { name = toe_pai_dale_agri value = 0 }
	set_variable = { name = toe_pai_dale_food value = 0 }
	set_variable = { name = toe_pai_shaftar_mining value = 0 }
	set_variable = { name = toe_pai_shaftar_tools value = 0 }
	set_variable = { name = toe_pai_eskan_plantation value = 0 }
	set_variable = { name = toe_pai_eskan_industry value = 0 }
	set_variable = { name = toe_pai_old_industry value = 0 }
	set_variable = { name = toe_pai_old_mining value = 0 }
	set_variable = { name = toe_pai_northern_agri value = 0 }
	set_variable = { name = toe_pai_palun_gold value = 0 }
	set_variable = { name = toe_pai_palun_mining value = 0 }
	# 非必須
	set_variable = { name = toe_pai_resind_shipyard value = 0 }
	set_variable = { name = toe_pai_prune_plantation value = 0 }
	set_variable = { name = toe_pai_krikul_ranch value = 0 }
	set_variable = { name = toe_pai_balin_arms value = 0 }
	set_variable = { name = toe_pai_balin_industry value = 0 }

	every_scope_state = {
		# デール
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_dale } }
			every_scope_building = {
				if = { limit = { OR = { is_building_type = building_wheat_farm is_building_type = building_rye_farm is_building_type = building_rice_farm is_building_type = building_maize_farm is_building_type = building_millet_farm } } ROOT = { change_variable = { name = toe_pai_dale_agri add = PREV.level } } }
				if = { limit = { is_building_type = building_food_industry } ROOT = { change_variable = { name = toe_pai_dale_food add = PREV.level } } }
			}
		}
		# シャフタパイド
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_shaftar } }
			every_scope_building = {
				if = { limit = { is_building_group = bg_mining } ROOT = { change_variable = { name = toe_pai_shaftar_mining add = PREV.level } } }
				if = { limit = { is_building_type = building_tooling_workshops } ROOT = { change_variable = { name = toe_pai_shaftar_tools add = PREV.level } } }
			}
		}
		# エスカン
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_eskan } }
			every_scope_building = {
				if = { limit = { is_building_group = bg_plantations } ROOT = { change_variable = { name = toe_pai_eskan_plantation add = PREV.level } } }
				if = { limit = { is_building_group = bg_manufacturing } ROOT = { change_variable = { name = toe_pai_eskan_industry add = PREV.level } } }
			}
		}
		# オールドパイド
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_old } }
			every_scope_building = {
				if = { limit = { is_building_group = bg_manufacturing } ROOT = { change_variable = { name = toe_pai_old_industry add = PREV.level } } }
				if = { limit = { is_building_group = bg_mining } ROOT = { change_variable = { name = toe_pai_old_mining add = PREV.level } } }
			}
		}
		# ノーザンパイド
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_northern } }
			every_scope_building = {
				if = { limit = { OR = { is_building_type = building_wheat_farm is_building_type = building_rye_farm is_building_type = building_rice_farm is_building_type = building_maize_farm is_building_type = building_millet_farm } } ROOT = { change_variable = { name = toe_pai_northern_agri add = PREV.level } } }
			}
		}
		# パルン
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_palun } }
			every_scope_building = {
				if = { limit = { is_building_type = building_gold_mine } ROOT = { change_variable = { name = toe_pai_palun_gold add = PREV.level } } }
				if = { limit = { is_building_group = bg_mining } ROOT = { change_variable = { name = toe_pai_palun_mining add = PREV.level } } }
			}
		}
		# レシンド
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_3 } }
			every_scope_building = { if = { limit = { is_building_type = building_shipyards } ROOT = { change_variable = { name = toe_pai_resind_shipyard add = PREV.level } } } }
		}
		# プルーン
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_5 } }
			every_scope_building = { if = { limit = { is_building_group = bg_plantations } ROOT = { change_variable = { name = toe_pai_prune_plantation add = PREV.level } } } }
		}
		# クリクル
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_2 } }
			every_scope_building = { if = { limit = { is_building_type = building_livestock_ranch } ROOT = { change_variable = { name = toe_pai_krikul_ranch add = PREV.level } } } }
		}
		# バリン
		if = {
			limit = { state_region = { is_homeland = cu:nsm_map_pai_4 } }
			every_scope_building = {
				if = { limit = { is_building_type = building_arms_industry } ROOT = { change_variable = { name = toe_pai_balin_arms add = PREV.level } } }
				if = { limit = { is_building_group = bg_manufacturing } ROOT = { change_variable = { name = toe_pai_balin_industry add = PREV.level } } }
			}
		}
	}

	# 条件達成を自動反映。
	if = { limit = { NOT = { has_variable = toe_paido_done_dale } var:toe_pai_dale_agri >= 50 var:toe_pai_dale_food >= 10 } toe_paido_integrate_dale = yes }
	if = { limit = { NOT = { has_variable = toe_paido_done_shaftar } var:toe_pai_shaftar_mining >= 20 var:toe_pai_shaftar_tools >= 5 } toe_paido_integrate_shaftar = yes }
	if = { limit = { NOT = { has_variable = toe_paido_done_eskan } var:toe_pai_eskan_plantation >= 20 var:toe_pai_eskan_industry >= 5 } toe_paido_integrate_eskan = yes }
	if = { limit = { NOT = { has_variable = toe_paido_done_old } var:toe_pai_old_industry >= 10 var:toe_pai_old_mining >= 5 } toe_paido_integrate_old = yes }
	if = { limit = { NOT = { has_variable = toe_paido_done_northern } var:toe_pai_northern_agri >= 15 } toe_paido_integrate_northern = yes }
	if = { limit = { NOT = { has_variable = toe_paido_done_palun } var:toe_pai_palun_gold >= 3 var:toe_pai_palun_mining >= 5 } toe_paido_integrate_palun = yes }

	if = { limit = { NOT = { has_variable = toe_paido_done_resind } var:toe_pai_resind_shipyard >= 5 country_average_culture_pop_acceptance = { culture = cu:nsm_map_pai_3 value >= 50 } } toe_paido_tradition_resind = yes }
	if = { limit = { NOT = { has_variable = toe_paido_done_prune } var:toe_pai_prune_plantation >= 10 country_average_culture_pop_acceptance = { culture = cu:nsm_map_pai_5 value >= 50 } } toe_paido_tradition_prune = yes }
	if = { limit = { NOT = { has_variable = toe_paido_done_krikul } var:toe_pai_krikul_ranch >= 15 country_average_culture_pop_acceptance = { culture = cu:nsm_map_pai_2 value >= 50 } } toe_paido_tradition_krikul = yes }
	if = { limit = { NOT = { has_variable = toe_paido_done_sosilia } c:SOS ?= { gdp >= 6000000 relations:root >= 50 } } toe_paido_tradition_sosilia = yes }
	if = { limit = { NOT = { has_variable = toe_paido_done_balin } var:toe_pai_balin_arms >= 5 var:toe_pai_balin_industry >= 5 } toe_paido_integrate_balin = yes }
}

# 成功報酬を達成数に合わせて選ぶ。
toe_paido_apply_integration_reward = {
	if = { limit = { var:toe_paido_achieved_count >= 11 } add_modifier = { name = toe_paido_people_11 days = 3650 } }
	else_if = { limit = { var:toe_paido_achieved_count >= 10 } add_modifier = { name = toe_paido_people_10 days = 3650 } }
	else_if = { limit = { var:toe_paido_achieved_count >= 9 } add_modifier = { name = toe_paido_people_9 days = 3650 } }
	else_if = { limit = { var:toe_paido_achieved_count >= 8 } add_modifier = { name = toe_paido_people_8 days = 3650 } }
	else_if = { limit = { var:toe_paido_achieved_count >= 7 } add_modifier = { name = toe_paido_people_7 days = 3650 } }
	else = { add_modifier = { name = toe_paido_people_6 days = 3650 } }
}
