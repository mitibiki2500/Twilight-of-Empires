    when_taken = {
        ig:ig_intelligentsia = { save_scope_as = toe_ria_pollution_fund_sponsor }

            add_amendment = {
                type = amendment_toe_ria_soil_pollution_fund
                sponsor = scope:toe_ria_pollution_fund_sponsor
            }
        }
        set_variable = toe_ria_soil_pollution_fund_adopted
    }

    ai_chance = { value = 25 }
