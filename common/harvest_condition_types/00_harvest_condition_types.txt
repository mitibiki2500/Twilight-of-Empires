# Twilight of Empires custom harvest conditions
# Great War chemical / biological / radiological compound contamination
toe_great_war_compound_contamination = {
	area_type = land
	has_animation = yes

	# This condition is only started by history or gas warfare scripts.
	trigger = { always = no }

	range = {
		integer_range = { min = 0 max = 0 }
	}

	# Default 25 years. Explicit starts may use shorter durations by intensity.
	duration = {
		fixed_range = { min = 9125 max = 9125 }
	}

	modifier = {
		building_group_bg_agriculture_throughput_add = -0.05
		building_group_bg_plantations_throughput_add = -0.05
		state_migration_pull_mult = -0.025
		state_birth_rate_mult = -0.02
		state_mortality_mult = 0.02
		state_standard_of_living_add = -0.1
	}

	intensity = {
		fixed_range = { min = 1 max = 1 }
	}

	chance = { value = 0 }

	color = { 120 86 45 }
	icon = "gfx/interface/icons/harvest_condition_icons/wildfire.dds"
	map_texture = "gfx/map/map_painting/harvest_conditions_map_patterns/harvest_conditions_bad.dds"
	graphics = disease_outbreak
}
