			text = mobilization_option_it_hurts_organization_when_you_adjust_equipment_tt

				value = organization

			}
		}
	}

		paratroopers
	}
	upkeep_modifier = {
		goods_input_aeroplanes_add = 0.5
		goods_input_fabric_add = 0.5
		goods_input_oil_add = 0.5
