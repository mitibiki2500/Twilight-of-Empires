force_become_subject = {
	groups= {
		power_bloc_leader
	}
	show_in_lens = no

	texture = "gfx/interface/icons/diplomatic_action_icons/force_become_subject.dds"

	selectable = {
		is_power_bloc_leader = yes
		power_bloc ?= {
			modifier:power_bloc_leader_can_make_subjects_bool = yes
		}
	}

	potential = {
		power_bloc ?= {
			scope:target_country.power_bloc ?= this
		}
	}

	possible = {
		scope:target_country = {
			is_subject = no
			country_rank < rank_value:unrecognized_major_power
			is_diplomatic_play_committed_participant = no
		}

		power_bloc = {
			current_cohesion_percentage >= 0.5
		}
		scope:target_country = {
			tenure_in_current_power_bloc_years >= 5
		}
		prestige >= {
			value = scope:target_country.prestige
			multiply = 5
		}
	}

	accept_effect = {
		if = {
			limit = { is_country_type = unrecognized }
			create_diplomatic_pact = {
				country = scope:target_country
				type = tributary
			}
		}
		else_if = {
			limit = {
				scope:target_country = {
					is_country_type = colonial
				}
			}
			create_diplomatic_pact = {
				country = scope:target_country
				type = dominion
			}
		}
		else = {
			create_diplomatic_pact = {
				country = scope:target_country
				type = protectorate
			}
		}
		create_unidirectional_truce = { country = scope:target_country months = 60 }
		hidden_effect = { scope:target_country = { add_liberty_desire = 30 } } # Starts at a higher LD level
		hidden_effect = {
			if = {
				limit = {
					has_journal_entry = je_aul_dwarov
				}
				set_variable = {
					name = temp_current_holds
					value = country_serpentspine_holds_count
				}
				if = {
					limit = {
						var:temp_current_holds < var:stored_holds
					}
					set_variable = aul_dwarov_hold_loss_flag
				}
				change_variable = {
					name = temp_current_holds
					subtract = var:stored_holds
				}
				je:je_aul_dwarov ?= {
					add_progress = { value = owner.var:temp_current_holds name = anb_aul_dwarov_holds_bar }
				}
				change_variable = {
					name = temp_current_holds
					multiply = 20
				}
				je:je_aul_dwarov ?= {
					add_progress = { value = owner.var:temp_current_holds name = anb_aul_dwarov_zeal_bar }
				}
				set_variable = {
					name = stored_holds
					value = country_serpentspine_holds_count
				}
			}
		}
		power_bloc = {
			add_cohesion_percent = {
				value = scaled_cohesion_cost_subjugation
			}
		}
	}

	ai = {
		evaluation_chance = {
			value = 0.025 # AI shouldn't use this very often
		}

		will_propose = {
			power_bloc ?= { current_cohesion_percentage >= 0.6 }
			OR = {
				has_attitude = {
					who = scope:target_country
					attitude = domineering
				}
				power_bloc ?= { current_cohesion_percentage >= 0.8 }
			}
			OR = {
				AND = {
					has_attitude = {
						who = scope:target_country
						attitude = domineering
					}
					power_bloc ?= { current_cohesion_percentage >= 0.8 }
				}
				scope:target_country = {
					is_country_type = unrecognized
				}
			}
		}

		propose_score = {
			value = 10
		}
	}
}
