alecand_reclamation_progress_bar = {
	name = "alecand_reclamation_progress_bar"
	desc = "alecand_reclamation_progress_bar_desc"

	default = yes

	monthly_progress = {
		if = {
			limit = {
				has_technology_researched = punch_card_artificery
			}
			add = {
				desc = "anb_reclamation_punch_card_artificery_tt"
				value = 1
			}
		}

		if = {
			limit = {
				has_technology_researched = elemental_elicitation
			}
			add = {
				desc = "anb_reclamation_elemental_elicitation_tt"
				value = 1
			}
		}

		if = {
			limit = {
				has_technology_researched = thaumic_field_stabilization
			}
			add = {
				desc = "anb_reclamation_thaumic_field_stabilization_tt"
				value = 1
			}
		}
		if = {
			limit = {
				has_variable = var_innovation_reclamation_button_spent_var
			}
			add = {
				desc = "anb_reclamation_buttons_innovation_tt"
				value = 1
			}
		}

		if = {
			limit = {
				has_variable = var_construction_reclamation_button_spent_var
			}
			add = {
				desc = "anb_reclamation_buttons_construction_tt"
				value = 2
			}
		}

		add = {
          	desc = "anb_reclamation_goods_tt"
          	market_capital.market = {
              	mg:artificery_doodads = {
                  	if = {
                      	limit = {
                          	market_goods_production >= 500
                      	}
                      	value = 1
                  		}
                  		else = {
                      	value = market_goods_production
                      	divide = 500
                  	}
              	}
          	}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 700
}

mechanim_awakening_progress_bar = {
	name = "mechanim_awakening_progress_bar"
	desc = "mechanim_awakening_progress_bar_desc"

	default_green = yes

	monthly_progress = {
		if = {
			limit = {
				exists = owner.var:mechanim_awakening_sentience_support_progress_tech
				owner.var:mechanim_awakening_sentience_support_progress_tech != 0 
			}
			add = {
				desc = "mechanim_awakening_sentience_support_progress_tech.tt"
				value = owner.var:mechanim_awakening_sentience_support_progress_tech
			}
		}
		if = {
			limit = {
				exists = owner.var:mechanim_awakening_sentience_support_progress_from_industrialists
				owner.var:mechanim_awakening_sentience_support_progress_from_industrialists != 0
			}
			add = {
				desc = "mechanim_awakening_sentience_support_progress_from_industrialists.tt"
				value = owner.var:mechanim_awakening_sentience_support_progress_from_industrialists
			}
		}
		if = {
			limit = {
				exists = owner.var:mechanim_awakening_sentience_support_progress_from_events
				owner.var:mechanim_awakening_sentience_support_progress_from_events != 0
			}
			add = {
				desc = "mechanim_awakening_sentience_support_progress_from_events.tt"
				value = owner.var:mechanim_awakening_sentience_support_progress_from_events
			}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 100
}
