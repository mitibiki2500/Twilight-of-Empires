movement_mechanim_liberation = {
	category = movement_category_ideological

	ideology = ideology_anti_mechanim_trade
	character_ideologies = {
		ideology_mechanim_liberator
	}
	
	creation_trigger = {
		NOT = { has_law = law_type:law_mechanim_freedom }
		has_technology_researched = mechanim_rights
		any_scope_state = {
			sg:automata = {
				state_goods_consumption > 1
			}
		}
	}	
	creation_weight = {
		value = 500
	}	
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}
	disband_trigger = {
		has_law = law_type:law_mechanim_freedom
	}
	
	character_support_trigger = {
		trigger_if = {
			limit = {
				is_in_exile_pool = yes
			}
			has_ideology = ideology:ideology_mechanim_liberator
		}
		trigger_else = {
			OR = {
				has_ideology = ideology:ideology_mechanim_liberator
				has_ideology = ideology:ideology_abolitionist
			}
			interest_group ?= {
				has_ideology = ideology:ideology_anti_mechanim_trade
			}
		}
	}
	character_support_weight = {
		value = 200
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_mechanim_liberator
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_abolitionist
				}
			}
			multiply = {
				value = 2
			}
		}
	}
	
	pop_support_trigger = {
		state = { 
			OR = {
				has_building = building_automatories
				sg:automata = {
					state_goods_consumption > 1
				}
			}
		}
	}

	pop_support_factors = {
		movement_support_automatories
		movement_support_high_literacy
		movement_support_machinists
		movement_support_academics
		movement_support_clerks
		movement_support_officers
		movement_support_engineers
		movement_support_high_urbanization
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
						OR = {
							is_being_suppressed = yes
							is_being_bolstered = yes
						}
					}
				}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}	
		add = {
			value = 14
			desc = "AUTOMATORIES_SUPPORT"
		}	

		if = {
			limit = {
				is_pop_type = machinists
			}
			add = {
				value = 150
				desc = "POP_MACHINISTS"
			}
		}	
		else_if = {
			limit = {
				is_pop_type = laborers
			}
			add = {
				value = 70
				desc = "POP_LABORERS"
			}
		}		
		else_if = {
			limit = {
				is_pop_type = clerks
			}
			add = {
				value = 22
				desc = "POP_CLERKS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = engineers
			}
			add = {
				value = 30
				desc = "POP_ENGINEERS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = academics
			}
			add = {
				value = 35
				desc = "POP_ACADEMICS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = officers
			}
			add = {
				value = 20
				desc = "POP_OFFICERS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = capitalists
			}
			add = {
				value = -14
				desc = "POP_CAPITALISTS"
			}
		}

		multiply = {  #Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}
		if = {
			limit = {
				owner = { NOT = { has_variable = mechanim_revolt_happened } }
			}
			multiply = {  #Mechanim liberation is a fringe issue until revolt happens
				desc = "NO_MECHANIM_REVOLT"
				value = 0.10
			}
		}
	}
	
	revolution = {
		possible = {
			political_movement_support >= 0.05
		}
			
		weight = {
			value = 100
		}

		state_weight = {
			value = 1
			
			if = {
				limit = { is_capital = yes }
				if = {
					limit = { owner ?= { NOT = { has_law = law_type:law_mechanim_freedom } } }
					multiply = 10.0
				}
				else = {
					multiply = 0.1
				}
			}		
		}
	}
	
	secession = {
		possible = {
			political_movement_support >= 0.05
		}	
	
		weight = {
			value = 1000
		}
		
		forced_tags = {
		}
		
		state_weight = {
			value = 1
			if = {
			   limit = { 
					NOT = {
						OR = {
							has_building = building_automatories
							sg:automata = {
								state_goods_consumption > 1
							}
						}
					}
			   }	
			   multiply = 0.0
			}
		}
	}
	additional_radicalism_factors = {
		if = {
			limit = {
				owner = {
					has_variable = mechanim_revolt_full_swing
				}
			}
			
			add = {
				value = 1.0
				desc = "MECHANIM_REVOLT_HAPPENING"
			}
		}
		if = {
			limit = {
				owner = {
					has_law = law_type:law_mechanim_as_property
				}
			}
			add = {
				value = 0.25
				desc = "law_mechanim_as_property"
			}
		}
	}
}
