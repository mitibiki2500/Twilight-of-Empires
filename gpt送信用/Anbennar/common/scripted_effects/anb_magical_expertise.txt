magical_expertise_apply_surplus_or_deficit_modifier = {
	### Applying surplus or deficit
	if = {
		limit = { has_modifier = country_magical_expertise_surplus_bonus_modifier }
		remove_modifier = country_magical_expertise_surplus_bonus_modifier
	}
	else_if = {
		limit = { has_modifier = country_magical_expertise_deficit_malus_modifier }
		remove_modifier = country_magical_expertise_deficit_malus_modifier
	}
	if = {
		limit = { 
            country_magical_expertise_balance >= 0
            country_num_spells > 0 
        }
		add_modifier = {
			name = country_magical_expertise_surplus_bonus_modifier
			multiplier = country_normalized_magical_expertise_surplus
			months = -1
		}
		if = { #AI check to increase spell levels
			limit = { is_ai = yes } 
			if = { #If has none of them upgrading AND has at least 1 spell level below lvl 5
				limit = {
					country_magical_expertise_balance >= country_magical_expertise_cost_per_institution #Check if they can actually afford it
					has_variable = production_magic_spell_level
					NOT = {
						OR = {
							has_variable = production_magic_spell_changing_level 
							has_variable = military_magic_spell_changing_level
							has_variable = society_magic_spell_changing_level
						}
					}
					OR = {
						var:production_magic_spell_level < 5
						var:military_magic_spell_level < 5
						var:society_magic_spell_level < 5
					}
				}
				random_list = {
					3 = {
						trigger = { var:production_magic_spell_level < 5 }
						set_variable = {
							name = production_magic_targeted_spell_level
							value = var:production_magic_spell_level
						}
						change_variable = {
							name = production_magic_targeted_spell_level
							add = 1
						}
						set_variable = { name = production_magic_spell_changing_level } 
					}
					3 = {
						trigger = { var:military_magic_spell_level < 5 }
						set_variable = {
							name = military_magic_targeted_spell_level
							value = var:military_magic_spell_level
						}
						change_variable = {
							name = military_magic_targeted_spell_level
							add = 1
						}
						set_variable = { name = military_magic_spell_changing_level } 
					}
					3 = {
						trigger = { var:society_magic_spell_level < 5 }
						set_variable = {
							name = society_magic_targeted_spell_level
							value = var:society_magic_spell_level
						}
						change_variable = {
							name = society_magic_targeted_spell_level
							add = 1
						}
						set_variable = { name = society_magic_spell_changing_level } 
					}
				}
			}
		}
	}
	else_if = {
		limit = { country_magical_expertise_balance < 0 }
		add_modifier = {
			name = country_magical_expertise_deficit_malus_modifier
			multiplier = country_normalized_magical_expertise_deficit
			months = -1
		}
		if = { #AI check to decrease spell levels
			limit = { is_ai = yes } 
			if = { #If has none of them downgrading AND has at least 1 spell level at lvl 1
				limit = {
					NOR = {
						has_variable = production_magic_spell_changing_level 
						has_variable = military_magic_spell_changing_level
						has_variable = society_magic_spell_changing_level
					}
					OR = {
						var:production_magic_spell_level >= 1
						var:military_magic_spell_level >= 1
						var:society_magic_spell_level >= 1
					}
				}
				random_list = {
					3 = {
						trigger = { var:production_magic_spell_level >= 1 }
						set_variable = {
							name = production_magic_targeted_spell_level
							value = var:production_magic_spell_level
						}
						change_variable = {
							name = production_magic_targeted_spell_level
							add = -1
						}
						set_variable = { name = production_magic_spell_changing_level } 
					}
					3 = {
						trigger = { var:military_magic_spell_level >= 1 }
						set_variable = {
							name = military_magic_targeted_spell_level
							value = var:military_magic_spell_level
						}
						change_variable = {
							name = military_magic_targeted_spell_level
							add = -1
						}
						set_variable = { name = military_magic_spell_changing_level } 
					}
					3 = {
						trigger = { var:society_magic_spell_level >= 1 }
						set_variable = {
							name = society_magic_targeted_spell_level
							value = var:society_magic_spell_level
						}
						change_variable = {
							name = society_magic_targeted_spell_level
							add = -1
						}
						set_variable = { name = society_magic_spell_changing_level } 
					}
				}
			}
		}
	}
	### Applying changes in magical expertise based on spell progression
	## Production
	# Raise the spell level if target level is higher + has required expertise
	if = { 
		# If changing production spell level, check to see what's up
		limit = { has_variable = production_magic_spell_changing_level }
		if = {
			limit = {
				var:production_magic_spell_level < var:production_magic_targeted_spell_level 
				country_difference_before_next_production_magic_level < country_affordable_spell_increases 
			}
			change_variable = {
				name = production_magic_spell_level
				add = 0.1 # 40 weeks (20 months) per level
				# note that i want this to be 50 weeks (12.5 months) per level, like institutions;
				# BUT it'd have to be a JE for that, for weekly pulses
			}
		}
		# Lower the spell level if target level is lower
		else_if = {
			limit = { var:production_magic_spell_level > var:production_magic_targeted_spell_level }
			change_variable = {
				name = production_magic_spell_level
				add = -0.1 # 40 weeks (10 months) per level, like institutions
			}
		}
		# If the calculated spell level has hit a break point, ensure the level is in the right place
		if = {
			limit = { country_modulo_production_magic_level = 0 }
			set_variable = {
				name = production_magic_spell_level_floor
				value = var:production_magic_spell_level
			}
			if = {
				limit = { var:production_magic_spell_level = var:production_magic_targeted_spell_level }
				remove_variable = production_magic_spell_changing_level
			}
			update_production_spell = yes
		}
	}
	## Military
    # Raise the spell level if target level is higher + has required expertise
    if = { 
        # If changing military spell level, check to see what's up
        limit = { has_variable = military_magic_spell_changing_level }
        if = {
            limit = {
                AND = {
                    var:military_magic_spell_level < var:military_magic_targeted_spell_level 
                    country_difference_before_next_military_magic_level < country_affordable_spell_increases 
                }
            }
            change_variable = {
                name = military_magic_spell_level
                add = 0.1 # 40 weeks (10 months) per level
                # note that i want this to be 50 weeks (12.5 months) per level, like institutions;
                # BUT it'd have to be a JE for that, for weekly pulses
            }
        }
        # Lower the spell level if target level is lower
        else_if = {
            limit = { var:military_magic_spell_level > var:military_magic_targeted_spell_level }
            change_variable = {
                name = military_magic_spell_level
                add = -0.1 # 40 weeks (10 months) per level, like institutions
            }
        }
        # If the calculated spell level has hit a break point, ensure the level is in the right place
        if = {
            limit = { country_modulo_military_magic_level = 0 }
            set_variable = {
                name = military_magic_spell_level_floor
                value = var:military_magic_spell_level
            }
            if = {
                limit = { var:military_magic_spell_level = var:military_magic_targeted_spell_level }
                remove_variable = military_magic_spell_changing_level
            }
			update_military_spell = yes
        }
    }	
	 ## Society
    # Raise the spell level if target level is higher + has required expertise
    if = { 
        # If changing society spell level, check to see what's up
        limit = { has_variable = society_magic_spell_changing_level }
        if = {
            limit = {
                AND = {
                    var:society_magic_spell_level < var:society_magic_targeted_spell_level 
                    country_difference_before_next_society_magic_level < country_affordable_spell_increases 
                }
            }
            change_variable = {
                name = society_magic_spell_level
                add = 0.1 # 40 weeks (10 months) per level
                # note that i want this to be 50 weeks (12.5 months) per level, like institutions;
                # BUT it'd have to be a JE for that, for weekly pulses
            }
        }
        # Lower the spell level if target level is lower
        else_if = {
            limit = { var:society_magic_spell_level > var:society_magic_targeted_spell_level }
            change_variable = {
                name = society_magic_spell_level
                add = -0.1 # 40 weeks (10 months) per level, like institutions
            }
        }
        # If the calculated spell level has hit a break point, ensure the level is in the right place
        if = {
            limit = { country_modulo_society_magic_level = 0 }
            set_variable = {
                name = society_magic_spell_level_floor
                value = var:society_magic_spell_level
            }
            if = {
                limit = { var:society_magic_spell_level = var:society_magic_targeted_spell_level }
                remove_variable = society_magic_spell_changing_level
            }
			update_society_spell = yes
        }
    }
}

		limit = {
			has_variable = production_magic_spell
			var:production_magic_spell_level_floor = 1
		}
		disable_production_spell = yes
	}
	else_if = {
		limit = {has_variable = production_magic_spell}
		change_variable = {
					name = production_magic_spell_level_floor
					subtract = 1
				}
		update_production_spell = yes
		if = {
			limit = { NOT = { has_variable = production_magic_spell_changing_level } }
			set_variable = {
				name = production_magic_targeted_spell_level
				value = var:production_magic_spell_level_floor

		}
	}
	#magical_expertise_apply_surplus_or_deficit_modifier = yes
}
