combat_unit_type_undead_infantry = {
    group = combat_unit_group_infantry
    max_manpower = 1000
	supply_capacity 10 
    conscript_peasant_levies = no

    battle_modifier = {
        unit_offense_add = 5
        unit_defense_add = 10
        unit_morale_loss_add = 0
        unit_recovery_rate_add = 1
        ### dev documentation for undead
        ### these guys require investing into the spell to get some decent values (equal to line infantery at max)
        ### recovery rate above 1 doesn't do anything whack. without further modifiers this means the units don't die but become demoralized in battle by being "killed"
        ### not touching barrack pms to keep it simple, you have to pay "wages" (upkeep) and with enemy modifiers or bad conditions/generals you may still lose some pops, 
        ### these can be written off as your logistic corps/wizards being shelled or you needing to import some fresh bodies from time to time because your enemy pulverized your units
        ### unit is currently for 2 tags and the AI is too stupid to use them, will not care for 1.0 about that. they're really fun to use now, bring exactly the undead army vibe they should and are op early and just become (probably) bad compared to regular infantry as mobilization options and stuff are unlocked.
    }
	
    upkeep_modifier = {
		goods_input_magical_reagents_add = 3
		goods_input_meat_add = 1
        country_magical_expertise_cost_add = 5
        
    }

	
	unlocking_technologies = { necromantic_magical_tradition }

	combat_unit_image = {
        texture = "gfx/unit_illustrations/infantry_undead.dds"
    }

}
