
portnamm_extracted_wealth_script_value = {
	every_country = {
		limit = { this != root }
		if = {
			limit = { "this.gdp_ownership_ratio(root)" > 0 }
			add = this.portnamm_one_foreign_gdp_helper_script_value
		}
	}
	divide = 2000000
}

portnamm_one_foreign_gdp_helper_script_value = {
	add = "gdp_ownership_ratio(root)"
	multiply = gdp
}


# control level is percentage of gdp controlled, but with penalty if total amount < 50M, used by JE5
level_of_control_by_portnamm_script_value = {
	add = "gdp_ownership_ratio(root)"
	multiply = controlled_gdp_size_scale_script_value
}


# A multiplier, (portnamm countrolled gdp in this country)/50M, with maximum 1, used by JE5
controlled_gdp_size_scale_script_value = {
	add = "gdp_ownership_ratio(root)"
	multiply = gdp
	divide = 50000000
	max = 1
}
