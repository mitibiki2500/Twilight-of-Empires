country_red_season_name_dynamic = {
	type = country
	text = {
		trigger = {
			OR = {
				month >= 11
				month < 2
			}
		}
		localization_key = red_winter_loc
	}
	text = {
		trigger = {
			month >= 2
			month < 5
		}
		localization_key = red_spring_loc
	}
	text = {
		trigger = {
			month >= 5
			month < 8
		}
		localization_key = red_summer_loc
	}
	text = {
		trigger = {
			month >= 8
			month < 11
		}
		localization_key = red_autumn_loc
	}
	text = { # Fallback
		trigger = {
			always = yes
		}
		localization_key = red_summer_loc
	}
}
