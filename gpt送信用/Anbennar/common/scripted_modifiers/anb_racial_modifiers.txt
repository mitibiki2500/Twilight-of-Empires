pop_high_profession_affinity = {
	#Can add stronger/weaker racial affinities here
	if = {
		limit = {
			this.culture = {
				is_$RACE$ = yes
			}
		}
		multiply = {
			value = 3
			desc = "QUALIFICATIONS_RACE_FAVORED_TYPE"
		}
	}
}
