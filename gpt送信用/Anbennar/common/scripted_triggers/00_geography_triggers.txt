is_arid_region = { # anb adjusted
	OR = {
		state_region ?= s:STATE_NORTHERN_SANDSPITE
		state_region ?= s:STATE_SOUTHERN_SANDSPITE
		state_region ?= s:STATE_WESTERN_SALAHAD
		state_region ?= s:STATE_CENTRAL_SALAHAD
		state_region ?= s:STATE_DEMON_SANDS
		state_region ?= s:STATE_ARDU
		state_region ?= s:STATE_KERUHAR
		state_region ?= s:STATE_FAR_EAST_SALAHAD
		region ?= sr:region_far_bulwar
		region ?= sr:region_fangaula
		region ?= sr:region_jinnakah
	}
}

is_country_on_same_continent = { # anb adjusted
	custom_description = {
		text = "is_country_on_same_continent"
		subject = this
		object = $TARGET$

		trigger_if = {
			limit = {
				this.capital.state_region = {
					is_in_geographic_region = geographic_region_cannor
				}
			}
			$TARGET$.capital.state_region = {
				is_in_geographic_region = geographic_region_cannor
			}
		}
		trigger_else_if = {
			limit = {
				this.capital.state_region = {
					is_in_geographic_region = geographic_region_north_aelantir
				}
			}
			$TARGET$.capital.state_region = {
				is_in_geographic_region = geographic_region_north_aelantir
			}
		}
		trigger_else_if = {
			limit = {
				this.capital.state_region = {
					is_in_geographic_region = geographic_region_south_aelantir
				}
			}
			$TARGET$.capital.state_region = {
				is_in_geographic_region = geographic_region_south_aelantir
			}
		}
		trigger_else_if = {
			limit = {
				this.capital.state_region = {
					is_in_geographic_region = geographic_region_haless
				}
			}
			$TARGET$.capital.state_region = {
				is_in_geographic_region = geographic_region_haless
			}
		}
		trigger_else_if = {
			limit = {
				this.capital.state_region = {
					is_in_geographic_region = geographic_region_sarhal
				}
			}
			$TARGET$.capital.state_region = {
				is_in_geographic_region = geographic_region_sarhal
			}
		}
		trigger_else_if = {
			limit = {
				this.capital.state_region = {
					is_in_geographic_region = geographic_region_dwarovar
				}
			}
			$TARGET$.capital.state_region = {
				is_in_geographic_region = geographic_region_dwarovar
			}
		}
		trigger_else_if = {
			limit = {
				this.capital.state_region = {
					is_in_geographic_region = geographic_region_unreleased
				}
			}
			$TARGET$.capital.state_region = {
				is_in_geographic_region = geographic_region_unreleased
			}
		}
	}
}
