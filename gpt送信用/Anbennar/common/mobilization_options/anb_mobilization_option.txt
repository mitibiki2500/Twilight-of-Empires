	texture = "gfx/interface/icons/mobilization_options/blackdamestear.dds"

	possible = {
		market ?= {
			mg:artificery_doodads ?= {
				has_potential_supply = yes
			}
		}
		
	}

			}
		}

				value = organization

			}
		}
	}
	
	unlocking_technologies = {
		elemental_armaments
	}
	upkeep_modifier = {
		goods_input_damestear_add = 0.5

	}
	unit_modifier = {
		unit_morale_damage_mult = 0.2
	}
	ai_weight = {
		value = 1
	}

    group = infantry_weapons
}

mobilization_option_gene_warriors = {
	texture = "gfx/interface/icons/mobilization_options/artificerstuff.dds"

	unlocking_technologies = {
		gene_warriors
	}
	possible = {
		market ?= {
			mg:opium ?= {
				has_potential_supply = yes
			}
			mg:artificery_doodads ?= {
				has_potential_supply = yes
			}
			mg:fertilizer ?= {
				has_potential_supply = yes
			}
		}
	}
	on_activate_while_mobilized = {
		custom_tooltip = {
			text = mobilization_option_it_hurts_organization_to_add_medic_support_tt
			add_organization = {
				value = organization
				multiply = -0.25
			}
		}
	}
	
	on_deactivate = {
		custom_tooltip = {
			text = mobilization_option_it_hurts_morale_and_organization_when_you_remove_medic_support_tt
			every_combat_unit = {
				add_morale = {
					value = morale
					multiply = -0.5
				}
			}
			add_organization = {
				value = organization
				multiply = -0.25
			}
		}
	}
	upkeep_modifier = {
		goods_input_opium_add = 4
		goods_input_artificery_doodads_add = 2
		goods_input_fertilizer_add = 4
	}
	unit_modifier = {
		unit_offense_mult = 0.3	#broken on purpose as these are just supersoldiers tbh. its a multiplier as you do need to give them equipment, which is everything else
		unit_defense_mult = 0.3
	}
	ai_weight = {
		value = 1
	}

    group = medic_support
}
