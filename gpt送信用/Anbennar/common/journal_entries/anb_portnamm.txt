	}

	on_monthly_pulse = {
		effect = {

				name = portnamm_age_of_art_active_modifier
				multiplier = "scope:journal_entry.scripted_bar_progress(portnamm_age_of_art_bar)"
				is_decaying = no
				months = -1
			}
		}
	}

	complete = {
		always = no
	}

	fail = {
		scope:journal_entry = {
			custom_tooltip = {
				text = portnamm_age_of_art_tt
				"scripted_bar_progress(portnamm_age_of_art_bar)" <= 0
			}
		}
	}

		add_modifier = {
			name = portnamm_industrialists_ascendancy
			years = 20
			is_decaying = yes
		}
		remove_modifier = portnamm_age_of_art_active_modifier
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}


### JE1: Rise of Gnome Industrialists ###
je_portnamm_rise_of_industrialists = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	group = je_group_historical_content

	immediate = {
		set_variable = {
			name = portnamm_activated_directorate_companies
			value = 0
		}
		set_variable = {
			name = portnamm_directorate_companies_allowed
			value = 0
		}
	}

	modifiers_while_active = {
		portnamm_je1_active_modifier
	}

	scripted_button = portnamm_je1_lorent_pact_button

	complete = {
		ig:ig_industrialists = {
			ig_clout >= 0.20
		}
		custom_tooltip = {
			text = portnamm_je1_complete_tt
			has_variable = portnamm_lorent_pact_active
		}
	}

	on_complete = {
		# Remove Individualist and Laissez-Faire from industrialist IG
		# Add Directoratism and Vaultarchy
		ig:ig_industrialists = {
			remove_ideology = ideology_individualist
			remove_ideology = ideology_laissez_faire
			add_ideology = ideology_directoratism
			add_ideology = ideology_vaultarchy
		}

		# Activate JE2
		add_journal_entry = { type = je_portnamm_sovereign_ledger }
		add_journal_entry = { type = je_portnamm_directoratism_reforms }
	}

	fail = {
		OR = {
			NOT = {
				is_in_same_power_bloc = c:A03
			}
			c:A03 = { NOT = { country_rank = rank_value:great_power } }
			custom_tooltip = {
				text = portnamm_je1_fail_more_than_one_state_tt
				any_state = {
					count >= 2
					owner = c:A15
				}
			}
			custom_tooltip = {
				text = portnamm_je1_tech_requirement_tt
				c:A03 = { has_technology_researched = insyaan_analytical_engine }
				c:A03 = { has_technology_researched = fractional_distillation }
				c:A03 = { has_technology_researched = crystal_glass }
			}
		}
	}

	on_fail = {
		trigger_event = { id = anb_portnamm_events.1 }
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}


### JE2: The Sovereign Ledger Directorate ###
je_portnamm_sovereign_ledger = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	group = je_group_historical_content

	modifiers_while_active = {
		portnamm_je2_active_modifier
	}

	immediate = {
		# Add construction and prestige modifier (scaled by wealth)
		add_modifier = {
			name = portnamm_wealth_extraction_impact
			multiplier = portnamm_extracted_wealth_script_value
			is_decaying = no
			months = -1
		}
		set_variable = portnamm_directorate_active
	}


	on_monthly_pulse = {
		effect = {
			remove_modifier = portnamm_wealth_extraction_impact
			add_modifier = {
				name = portnamm_wealth_extraction_impact
				multiplier = portnamm_extracted_wealth_script_value
				is_decaying = no
				months = -1
			}

			# Calculate how many directorate companies are allowed
			# 1 at 10M, 2 at 30M, 3 at 60M, 4 at 150M, 5 at 300M, 6 at 500M
			if = {
				limit = { portnamm_extracted_wealth_script_value < 5 }
				change_variable = { name = portnamm_directorate_companies_allowed max = 0 }
			}
			else_if = {
				limit = { 
					portnamm_extracted_wealth_script_value >= 5
					portnamm_extracted_wealth_script_value < 15
				}
				change_variable = { name = portnamm_directorate_companies_allowed max = 1 }
			}
			else_if = {
				limit = { 
					portnamm_extracted_wealth_script_value >= 15
					portnamm_extracted_wealth_script_value < 30
				}
				change_variable = { name = portnamm_directorate_companies_allowed max = 2 }
			}
			else_if = {
				limit = { 
					portnamm_extracted_wealth_script_value >= 30
					portnamm_extracted_wealth_script_value < 75
				}
				change_variable = { name = portnamm_directorate_companies_allowed max = 3 }
			}
			else_if = {
				limit = { 
					portnamm_extracted_wealth_script_value >= 75
					portnamm_extracted_wealth_script_value < 150
				}
				change_variable = { name = portnamm_directorate_companies_allowed max = 4 }
			}
			else_if = {
				limit = { 
					portnamm_extracted_wealth_script_value >= 150
				}
				change_variable = { name = portnamm_directorate_companies_allowed max = 5 }
			}
		}
	}

	complete = {
		average_sol >= 35
		custom_tooltip = {
			text = portnamm_je2_complete_wealth_tt
			portnamm_extracted_wealth_script_value >= 500
		}
	}

	on_complete = {
		# Permanent bonuses
		add_modifier = {
			name = portnamm_je2_permanent_modifier
			is_decaying = no
			months = -1
		}

		# Keep construction and prestige modifiers (they're already applied)
		# set_variable = portnamm_je2_complete

		# Congratulation event
		trigger_event = { id = anb_portnamm_events.3 }
	}

	fail = {
		custom_tooltip = {
			text = portnamm_je2_fail_laws_tt
			OR = {
				has_variable = portnamm_je3_timed_out
				has_variable = portnamm_je4_fail
			}
		}
	}

	on_fail = {
		remove_modifier = portnamm_wealth_extraction_impact
		remove_variable = portnamm_directorate_active
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
