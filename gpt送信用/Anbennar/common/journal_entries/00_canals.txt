je_dhal_survey = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_foreign_affairs

	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = survey_dhal_nikhuvad_taken
				add = 1
			}
		}
	}

	complete = {
		scope:journal_entry = { is_goal_complete = yes }
	}

	on_complete = {
		remove_modifier = modifier_surveying_dhal_nikhuvad
		trigger_event = { id = canal_events.100 }
	}

	current_value = {
		value = root.var:survey_dhal_nikhuvad_taken
	}

	goal_add_value = {
		value = 10
	}

	invalid = {
		OR = {
			NOT = { has_variable = survey_dhal_nikhuvad_taken }
			s:STATE_BEERAGGA = {
				any_scope_state = {
					NOT = { owner = ROOT }
					has_building = building_dhal_nikhuvad_canal
				}
			}
		}
	}

	on_invalid = {
		remove_variable = survey_dhal_nikhuvad_taken
		remove_modifier = modifier_surveying_dhal_nikhuvad
	}

	progressbar = yes

	weight = 10
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_dhal_nikhuvad_canal = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_technology

	complete = {
		s:STATE_BEERAGGA = {
			any_scope_state = {
				owner = root
				any_scope_building = {
					is_building_type = building_dhal_nikhuvad_canal
					occupancy >= 0.9
				}
			}
		}
	}

	on_complete = {
		set_global_variable = dhal_nikhuvad_canal_complete
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
}
