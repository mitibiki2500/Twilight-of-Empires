		}

				}

			}

			add_radicals = {
				value = very_large_radicals

			}
		}
	}

	on_monthly_pulse = {
		random_events = {
			200 = 0

			10 = northern_thaw_events.4
			10 = northern_thaw_events.5
		}
	}

	on_yearly_pulse = {
		random_events = {
			40 = 0
			10 = northern_thaw_events.6
		}
	}


	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_northern_thaw_woodswaller = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_nl_league_unity

	group = je_group_objectives #idk

	is_shown_when_inactive = {
		exists = c:A04
		THIS = c:A04
		any_interest_group = {
			leader = {
				has_template = A04_Tomac_Woodswall	#could be checking for the luddite ideology to be a bit more generic in case another luddite replaces him
			}
		}
	}

	scripted_button = northern_thaw_remove_woodwallers_button	#requires lvl 3 police

	#note : old content suggested some kind of interaction if Reach states get urbanized

	possible = {
		custom_tooltip = {
			text = nl_tomac_woodwaller_removed_tt
			NOT = {
				any_interest_group = {
					leader = {
						has_template = A04_Tomac_Woodswall	#could be checking for the luddite ideology to be a bit more generic in case another luddite replaces him
					}
				}
			}
		}
		NOR = {
			has_law = law_type:law_agrarianism
			has_law = law_type:law_industry_banned
			has_law = law_type:law_cooperative_ownership
			has_law = law_type:law_peasant_proprietorship
		}
	}

	immediate = {
		trigger_event = { id = woodswaller_rebellion_events.1 } 
		random_scope_state = {
			limit = {
				state_region = s:STATE_GREATWOODS
			}
			add_modifier = {
				name = nl_woodswaller_stronghold
				days = long_modifier_time
			}
		}
		every_scope_state = {
			limit = {
				NOT = {
					state_region = s:STATE_GREATWOODS
				}
				has_state_trait = state_trait_greatwoods
			}
			add_modifier = {
				name = nl_woodswaller_stronghold_small
				days = long_modifier_time
			}
		}
	}

	complete = {
		NOR = {
			any_scope_state = {
				has_modifier = nl_woodswaller_stronghold_small
			}
			any_scope_state = {
				has_modifier = nl_woodswaller_stronghold
			}
		}
	}

	on_complete = {
		nl_unity_increase_50 = yes
	}

	fail = {
		#still counts as done, concession ending
		OR = {
			has_law = law_type:law_agrarianism
			has_law = law_type:law_industry_banned
			has_law = law_type:law_cooperative_ownership
			# passing homesteading can appease them enough to prevent it, but once they revolt they want stronger concessions
		}
	}

	on_fail = {
		every_scope_state = {
			limit = {
				has_modifier = nl_woodswaller_stronghold_small
			}
			remove_modifier = nl_woodswaller_stronghold_small
		}
		every_scope_state = {
			limit = {
				has_modifier = nl_woodswaller_stronghold
			}
			remove_modifier = nl_woodswaller_stronghold
		}
	}

	on_monthly_pulse = {
		random_events = {
			200 = 0
			20 = woodswaller_rebellion_events.2		#change the frequency of that one to balance how annoying Woodswallers are
			20 = woodswaller_rebellion_events.4		#change the frequency of that one to balance Brasslance
			20 = woodswaller_rebellion_events.5
			5 = woodswaller_rebellion_events.7
			#TODO : depending on the shape AW stuff takes, add an event that spawns werewolves infestations
		}
	}

	on_yearly_pulse = {
		random_events = {
			40 = 0
			20 = woodswaller_rebellion_events.3	
			10 = woodswaller_rebellion_events.6	
		}
	}

	should_be_pinned_by_default_uninvolved_or_context = yes
}

#basic Reachmen nationalism JE for their side of the content
je_extent_of_our_reach = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_historical_content

	is_shown_when_inactive = {
		OR = {
			c:A18 ?= this
			c:A36 ?= this
			c:A39 ?= this
		}
	}

	possible = {
        has_technology_researched = nationalism
    }

	immediate = {
		trigger_event = { id = reach_events.1 }
	}

	complete = {
		any_scope_state = {
			state_region = {
				is_homeland = cu:blue_reachman
			}
			count >= 3
		}
		any_scope_state = {
			state_region = {
				is_homeland = cu:white_reachman
			}
			count >= 3
		}
	}

    on_complete = {
        if = {
			limit = {
				country_has_primary_culture = cu:blue_reachman
			}
			add_primary_culture = cu:white_reachman
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:white_reachman
			}
			add_primary_culture = cu:blue_reachman
		}
    }

    should_be_pinned_by_default_uninvolved_or_context = yes
}
