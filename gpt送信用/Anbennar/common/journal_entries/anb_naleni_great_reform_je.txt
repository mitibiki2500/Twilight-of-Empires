je_naleni_great_reform_main = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

    group = je_group_naleni_great_reform

    is_shown_in_lobby = {
        OR = {
            c:L29 ?= this
			was_formed_from = L29
        }
   	}

    is_shown_when_inactive = {
        OR = {
            c:L29 ?= this
			was_formed_from = L29
        }
    }

    timeout = 5475

    possible = {
        has_technology_researched = railways
        has_technology_researched = intensive_agriculture
        has_technology_researched = academia
    }

    immediate = {
        set_variable = {
            name = naleni_great_reform_var
            value = 0
        }
        set_variable = {
            name = naleni_great_reform_goal_var
            value = 5
        }
        random_scope_character = {
            limit = {
                has_template = naleni_jasiune_Nephiueda_template
            }
            save_scope_as = Jasiune_Nephiueda_scope
        }
        trigger_event = { id = naleni_great_reform_events.1 popup = yes }
    }

    can_deactivate = yes

    on_monthly_pulse = {
    }

    complete = {
        hidden_trigger = {
            exists = var:naleni_great_reform_goal_var
        }
        trigger_if = {
            limit = {
                exists = var:naleni_great_reform_var
            }
            custom_tooltip = {
                text = je_naleni_great_reform_master_completion_tt_2
                var:naleni_great_reform_var >= var:naleni_great_reform_goal_var
            }
        }
        NOT ={
        	has_law_or_variant = law_type:law_traditionalism
        }
    }

    event_outcome_completed_effect_desc = {
        header = naleni_great_reform_events.4.t
        effect = { #check modifiers
            add_modifier = {
                name = modifier_naleni_economic_reforms_sucess_reward
			    days = very_long_modifier_time
                is_decaying = yes
            }
            random_scope_character = {
                limit = {
                    has_template = naleni_jasiune_Nephiueda_template
                }
                add_modifier = {
                    name = modifier_great_reformer
                }
            }
            active_law:lawgroup_economic_system ?= {
                add_amendment = {
                    type = amendment_naleni_great_reform_1
                    sponsor = PREV.ig:ig_industrialists
                    cooldown = 240
                 }
            }
        }
    }

    on_complete = {
        set_variable = je_naleni_great_reform_main_completed_var
        trigger_event = {
            id = naleni_great_reform_events.4
            popup = yes
        }
    }

    on_timeout = {
        if = {
            limit = {
                exists = var:naleni_great_reform_var
            }
            if = {
                limit = {
                    var:naleni_great_reform_var >= 3
                }
                trigger_event = {
                    id = naleni_great_reform_events.6
                    popup = yes
                }
            }
        }
        else = {
            trigger_event = {
                id = naleni_great_reform_events.5
                popup = yes
            }
        }
    }

    on_fail = {
    }

    weight = 100

    should_be_pinned_by_default_uninvolved_or_context = yes
}
