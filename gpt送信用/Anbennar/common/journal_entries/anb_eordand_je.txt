	scripted_progress_bar = eordan_unification_bar

		any_scope_state = {	region = sr:region_eordand }
		NOT = { exists = c:B87 }

	}
	
	fail = {
		custom_tooltip = {
			text = eordan_unification_bar_at_full_tt

			}
		}

		every_country = {
			limit = {

		}
	}
	
	immediate = {

			trigger_event = { id = eordan_unification.4 }
		}
	}


	complete = {
		custom_tooltip = {
			text = eordan_unification_done_tt
			has_variable = eordan_unificator
		}
	}

	on_complete = {
		trigger_event = { id = eordan_unification.3 popup = yes }
	}
	
	invalid = {
		exists = c:B87
	}
	
	on_monthly_pulse = {
		effect = { 
			trigger_event = { id = eordan_unification.5 }
			scope:journal_entry = {
				set_bar_progress = {
					value = global_var:eordan_unfication_progress

	}

			}

		}
	}
