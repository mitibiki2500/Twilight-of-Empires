je_the_race_for_the_small_country_SC = { # Small Country fighting for it's independence against great powers
    # this is the main JE and the racers just a GP one to track progress
	icon = "gfx/interface/icons/event_icons/event_military.dds"	
	
	group = je_group_historical_content
    should_be_pinned_by_default_uninvolved_or_context = yes
    
    is_shown_when_inactive = {
		c:A14 ?= root # show only to Small Country when inactive
	}

    possible = { # activate if any of the following are true
        OR = {
            gdp_ownership_ratio = { target = c:A06 value >= 0.1 } # Gnomish Hierarchy owns greater than/equal 10% GDP of Small Country
            c:A06 = { has_technology_researched = nationalism } # Gnomish Hierarchy has researched Nationalism (they see halflings as distant cousins)
            calc_true_if = { # or if two or more of the key countries have a treaty with investment rights with Small Country
                amount >= 2
                c:A01 ?= { # Anbennar
                    any_scope_treaty = {
                        binds = root
                        any_scope_article = {
                            has_type = foreign_investment_rights
                        }
                    }
                }
                c:A03 ?= { # Lorent
                    any_scope_treaty = {
                        binds = root
                        any_scope_article = {
                            has_type = foreign_investment_rights
                        }
                    }
                }
                c:A04 ?= { # Northern League
                    any_scope_treaty = {
                        binds = root
                        any_scope_article = {
                            has_type = foreign_investment_rights
                        }
                    }
                }
                c:A06 ?= { # Gnomish Hierarchy
                    any_scope_treaty = {
                        binds = root
                        any_scope_article = {
                            has_type = foreign_investment_rights
                        }
                    }
                }
            }
        }
    }

    immediate = { # when activated do the following
        trigger_event = { id = anb_the_race_for_the_small_country.1 } # event for Small Country
        # Then send the other event to the following
        c:A01 = { # Anbennar
            trigger_event = { id = anb_the_race_for_the_small_country.2 }
        }
        c:A03 = { # Lorent
            trigger_event = { id = anb_the_race_for_the_small_country.2 }
        }
        c:A04 = { # Northern League
            trigger_event = { id = anb_the_race_for_the_small_country.2 }  
        }
        c:A06 = { # Gnomish Hierarchy
            trigger_event = { id = anb_the_race_for_the_small_country.2 }  
        }
    }

    complete = { # complete je when all four nations are either gone or have -100 leverage
        AND = {
            OR = {
                NOT = { exists = c:A01 }
                c:A01 ?= { # Anbennar
                    power_bloc = {
                        leverage_advantage = {
                            target = ROOT
                            value < -100
                        }
                    }
                }
            }
            OR = {
                NOT = { exists = c:A03 }
                c:A03 ?= { # Lorent
                    power_bloc = {
                        leverage_advantage = {
                            target = ROOT
                            value < -100
                        }
                    }
                }
            }
            OR = {
                NOT = { exists = c:A04 }
                c:A04 ?= { # Northern League
                    power_bloc = {
                        leverage_advantage = {
                            target = ROOT
                            value < -100
                        }
                    }
                }
            }
            OR = {
                NOT = { exists = c:A06 }
                c:A06 ?= { # Gnomish Hierarchy
                    power_bloc = {
                        leverage_advantage = {
                            target = ROOT
                            value < -100
                        }
                    }
                }
            }
        }
    }

    on_complete = { # send success event to Small Country
        trigger_event = { id = anb_the_race_for_the_small_country.3 }
    }
    
    timeout = 5475

 	fail = { # fail je if any of the four countries get 300 leverage on Small Country
        OR = {
            c:A01 ?= { # Anbennar
                power_bloc = {
                    leverage_advantage = {
                        target = ROOT
                        value > 300
                    }
                }
            }
            c:A03 ?= { # Lorent
                power_bloc = {
                    leverage_advantage = {
                        target = ROOT
                        value > 300
                    }
                }
            }
            c:A04 ?= { # Northern League
                power_bloc = {
                    leverage_advantage = {
                        target = ROOT
                        value > 300
                    }
                }
            }
            c:A06 ?= { # Gnomish Hierarchy
                power_bloc = {
                    leverage_advantage = {
                        target = ROOT
                        value > 300
                    }
                }
            }
        }
    }

 	on_fail = { # send success event to the country that gained 300 leverage
		if = {
            limit = {
                c:A01 ?= { # Anbennar
                    power_bloc = {
                        leverage_advantage = {
                            target = ROOT
                            value > 300
                        }
                    }
                }
            }
            c:A01 = {
                trigger_event = { id = anb_the_race_for_the_small_country.4 }  
            }
		}
		else_if = {
            limit = {
                c:A03 ?= { # Lorent
                    power_bloc = {
                        leverage_advantage = {
                            target = ROOT
                            value > 300
                        }
                    }
                }
            }
            c:A03 = {
                trigger_event = { id = anb_the_race_for_the_small_country.4 }  
            }
		}
		else_if = {
            limit = {
                c:A04 ?= { # Northern League
                    power_bloc = {
                        leverage_advantage = {
                            target = ROOT
                            value > 300
                        }
                    }
                }
            }
            c:A04 = {
                trigger_event = { id = anb_the_race_for_the_small_country.4 }  
            }
		}
		else_if = {
		    limit = {
                c:A06 ?= { # Gnomish Hierarchy
                    power_bloc = {
                        leverage_advantage = {
                            target = ROOT
                            value > 300
                        }
				    }
		        }
            }
            c:A06 = {
                trigger_event = { id = anb_the_race_for_the_small_country.5 } # Gnomish gets a unique event
            }
		}
    }
}
