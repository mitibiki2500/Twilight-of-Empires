law_military_bureaucrats = {
	group = lawgroup_bureaucracy
	
	icon = "gfx/interface/icons/law_icons/hereditary_bureaucrats.dds"
	
	progressiveness = -25

	#there is an argument that it should appointed
	#but right now hobgoblin AF and LO don't have any ideologies to change their flavor to be more about traditionalist military (like Bakufu or Caudillismo)
	#so enacting appointed can represent a switch from their old system to a proper one with dedicated bureaucrats for now
	parent = law_hereditary_bureaucrats
	
	modifier = {
		country_institution_cost_institution_police_mult = -0.25
		country_institution_size_change_speed_institution_police_mult = 0.25
		country_officers_pol_str_mult = 0.25
	}

	is_visible = {
		any_primary_culture = {
			OR = {
				has_discrimination_trait = society_eastern_wuhyun
				has_discrimination_trait = society_western_wuhyun
			}	#Command successors
		}
	}

	can_enact = {
		any_primary_culture = {
			OR = {
				has_discrimination_trait = society_eastern_wuhyun
				has_discrimination_trait = society_western_wuhyun
			}	#Command successors
		}
	}

	on_enact = {
		custom_tooltip = urban_aristocrats_int_ig_shift_effect_tt
		if = {
			limit = {
				has_law = law_type:law_elected_bureaucrats
			}
			custom_tooltip = bureaucrats_pb_ig_shift_effect_negative_tt
		}
	}

	disallowing_laws = {
		law_council_republic
		law_national_militia
	}

	ai_will_do = {
		exists = ruler
		ruler = {
			has_ideology = ideology:ideology_traditionalist
			has_ideology = ideology:ideology_jingoist
			has_ideology = ideology:ideology_fascist
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
