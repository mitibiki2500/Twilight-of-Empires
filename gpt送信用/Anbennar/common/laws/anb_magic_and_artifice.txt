law_artifice_banned = {
	group = lawgroup_magic_and_artifice
	
	icon = "gfx/interface/icons/law_icons/artifice_banned.dds"
	
	modifier = {
		country_magical_expertise_decree_cost_mult = -0.5
		country_archmages_pol_str_mult = 0.5
		country_mages_pol_str_mult = 0.5
		country_engineers_pol_str_mult = -0.5
		country_magical_expertise_mult = 0.5
	}
	
	on_enact = {
		recalculate_pop_ig_support = yes
		hidden_effect = { #accounts for if you had Magic Banned previously
			if = {
				limit = { 
                    AND = {
                        has_variable = magic_banned_var 
                        var:magic_banned_var = flag:magic_banned_flag 
                    }
                }
				remove_variable = magic_banned_var
				trigger_event = { id = anb_magical_expertise.2 days = 1 }
			}
		}
		if = {
			limit = { has_modifier = eordand_green_artifice_compensation }
			remove_modifier = eordand_green_artifice_compensation
		}
		custom_tooltip = {
			text = artifice_banned_destroy_tt
			every_scope_state = {
				if = {
					limit = {
						has_building = building_doodad_artificeries
					}
					remove_building = building_doodad_artificeries
				}
			}
		}
	}

	ai_enact_weight_modifier = {
		value = 0
	}

	progressiveness = -200
	
}

law_nation_of_green_artifice = {
	group = lawgroup_magic_and_artifice
	
	icon = "gfx/interface/icons/law_icons/consumption_based_taxation.dds"
	
	modifier = {
		country_magical_expertise_decree_cost_mult = -0.10
		country_tech_research_speed_mult = 0.025
		country_archmages_pol_str_mult = 0.10
		country_mages_pol_str_mult = 0.10
		country_engineers_pol_str_mult = 0.10
		state_pollution_reduction_health_mult = 1.0
		country_magical_expertise_mult = 0.125
	}
	
	is_visible = {
		has_variable = green_artifice_unlocked
	}
	
	can_enact = {
		has_variable = green_artifice_unlocked
	}
	
	on_enact = {
		recalculate_pop_ig_support = yes
		hidden_effect = { #accounts for if you had Magic Banned previously
			if = {
				limit = { 
                    AND = {
                        has_variable = magic_banned_var 
                        var:magic_banned_var = flag:magic_banned_flag 
                    }
                }
				remove_variable = magic_banned_var
				trigger_event = { id = anb_magical_expertise.2 days = 1 }
			}
		}
		if = {
			limit = { 
				OR = { 
					has_law = law_type:law_command_economy
					has_law = law_type:law_laissez_faire
				}
			}
			add_modifier = {	
				name = eordand_green_artifice_compensation
				multiplier = 2
			}
		}
		else_if = {
			limit = { 
				has_law = law_type:law_interventionism
			}
			add_modifier = {	
				name = eordand_green_artifice_compensation
				multiplier = 1
			}
		}
	}

	ai_enact_weight_modifier = {
		value = 0
	}

	progressiveness = 100
}

law_traditional_magic_banned = {
	group = lawgroup_magic_and_artifice
	
	icon = "gfx/interface/icons/law_icons/traditional_magic_banned.dds"
	
	disallowing_laws = {
		law_magocracy
	}
	
	
	modifier = {
		country_tech_research_speed_mult = 0.1
		country_archmages_pol_str_mult = -0.5
		country_mages_pol_str_mult = -0.5
		country_engineers_pol_str_mult = 0.5
		country_magical_expertise_mult = -1.0
	}
	
	on_enact = {
		recalculate_pop_ig_support = yes
		hidden_effect = {
			set_variable = { name = magic_banned_var value = flag:magic_banned_flag }
			initialize_spell_list = yes 
			disable_production_spell = yes
			disable_society_spell = yes
			disable_military_spell = yes
		}
		if = {
			limit = { has_modifier = eordand_green_artifice_compensation }
			remove_modifier = eordand_green_artifice_compensation
		}
		custom_tooltip = {
			text = magic_banned_destroy_tt
			every_scope_state = {
				if = {
					limit = {
						has_building = building_mage_tower
					}
					remove_building = building_mage_tower
				}
			}
		}
	}

	ai_enact_weight_modifier = {
		value = 0
	}

	progressiveness = 200
}
