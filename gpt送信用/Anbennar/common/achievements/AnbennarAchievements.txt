achievement_anb_machine_age = {
	possible = {
		always = yes ##Need to update once insyaa is done
	}
	happened = {
		has_law_or_variant = law_type:law_mechanim_freedom
		custom_tooltip = {
			text = lot_of_mechanim_tt
			any_scope_pop = {
				has_pop_culture = mechanim
				total_size >= 7500000
			}
		}
	}
}

achievement_anb_all_under_the_light = {
	possible = {
		has_variable = all_under_light_achievement_possible
	}
	happened = {
		has_law_or_variant = law_type:law_multicultural
		custom_tooltip = {
			text = two_hundred_jadd_pops_tt
			any_scope_pop = {
				has_pop_religion = the_jadd
				total_size >= 200000000
			}
		}
		custom_tooltip = {
			text = twenty_accepted_cultures_tt
			any_scope_culture = {
				NOT = {
					is_primary_culture_of = root
				}
				any_scope_pop = {
					filter = { state.owner = root }
					pop_acceptance >= acceptance_status_5
					percent >= 1
				}
				count <= 20
			}
		}
	}
}

achievement_anb_democracy = {
	possible = {
		has_variable = kalsyto_democracy_achievement_possible

					}

				count = all
			}
		}
	}
