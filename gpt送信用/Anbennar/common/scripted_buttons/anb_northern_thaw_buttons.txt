	desc = "northern_thaw_annex_cities_button_desc"

		value = 100
	}

		months = 6
	}

		always = yes

			}
			decrease_woodwaller_stronghold_size = yes
		}
	}
}

northern_thaw_annex_cities_button = {
	name = "northern_thaw_annex_cities_button"
	desc = "northern_thaw_annex_cities_button_desc"

	ai_chance = {	#AI does it that way to make sure the map gets cleaned up
		value = 10
	}

	visible = {
		NOT = {
			has_variable = nl_cities_integration_var
		}
		OR = {
			exists = c:A18
			exists = c:A19
		}
		AND = {
			c:A18 = {
				is_player = no
			}
			c:A19 = {
				is_player = no
			}
		}
	}

	possible = {
        has_technology_researched = nationalism
		OR = {
			c:A18 = {
				is_subject_of = ROOT
				is_player = no
			}
			c:A19 = {
				is_subject_of = ROOT
				is_player = no
			}
		}
	}

	effect = {
		set_variable = nl_cities_integration_var
		annex = c:A18
		annex = c:A19
		if = {
			limit = {
				exists = c:A01
			}
			create_incident = {
				country = c:A01
				value = 10
			}
		}
		add_radicals = {
			culture = cu:vertesker
			value = large_radicals
		}
		add_radicals = {
			culture = cu:city_goblin
			value = large_radicals
		}
	}
}

	name = "northern_thaw_conversion_focus_button"
	desc = "northern_thaw_conversion_focus_button_desc"

		value = 0

			limit = {

					has_strategy = ai_strategy_conservative_agenda

				}
			}

		}
	}

		}

			name = nl_recently_picked_ravelian_focus

			limit = {
				has_modifier = nl_encouraged_education
			}
			remove_modifier = nl_encouraged_education
			add_modifier = {
				name = nl_encouraged_proselytization
			}
		}
	}
}

northern_thaw_more_missionaries_button = {
	name = "northern_thaw_more_missionaries_button"
	desc = "northern_thaw_more_missionaries_button_desc"

	ai_chance = {
        value = 50
		if = {
			limit = {
				has_strategy = ai_strategy_egalitarian_agenda
			}
			subtract = 50
		}
    }

	cooldown = {
		days = short_modifier_time
	}

	visible = {
		NOT = {
			has_variable = nl_stopped_gerudia_conversion
		}
	}

	effect = {
		if = {	#try to pick one without modifiers yet
			limit = {
				any_scope_state = {
					state_region = {
						is_homeland = cu:dalr
					}
					is_incorporated = yes
					NOR = {
						has_modifier = nl_encouraged_proselytization
						has_modifier = nl_encouraged_education
					}
				}
			}
			random_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:dalr
					}
					is_incorporated = yes
					NOR = {
						has_modifier = nl_encouraged_proselytization
						has_modifier = nl_encouraged_education
					}
				}
				save_scope_as = new_mission_state
			}
		}
		else = {
			random_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:dalr
					}
					is_incorporated = yes
				}
				save_scope_as = new_mission_state
			}
		}

		scope:new_mission_state = {
			convert_population = {
				target = rel:ravelian
				value = 0.05
			} 
			if = {
				limit = {
					scope:root_country = {
						has_variable = nl_gerudia_conversion_focus
					}
					NOT = {
						has_modifier = nl_encouraged_proselytization
					}
				}
				add_modifier = {
					name = nl_encouraged_proselytization
				}
			}
			else_if = {
				limit = {
					scope:root_country = {
						has_variable = nl_gerudia_education_focus
					}
					NOT = {
						has_modifier = nl_encouraged_education
					}
				}
				add_modifier = {
					name = nl_encouraged_education
				}
			}
		}
		add_modifier = {
			name = nl_expanded_ravelian_mission
			days = short_modifier_time
		}
		add_modifier = {
			name = modifier_nl_conversion_costs
			days = short_modifier_time
		}
	}
}

northern_thaw_stop_missionaries_button = {
	name = "northern_thaw_stop_missionaries_button"
	desc = "northern_thaw_stop_missionaries_button_desc"

	ai_chance = {
		value = 0
		if = {
			limit = {
				has_strategy = ai_strategy_egalitarian_agenda
			}
			add = 50
		}
	}

	visible = {
		NOT = {
			has_variable = nl_stopped_gerudia_conversion
		}
	}

	possible = {
		ig:ig_devout = {
			is_in_government = no
		}
		any_interest_group = {
			is_in_government = yes
			leader = {
				has_ideology = ideology:ideology_nihilist
			}
		}
	}

	effect = {
		set_variable = nl_stopped_gerudia_conversion
		if = {
			limit = {
				has_variable = nl_gerudia_conversion_focus
			}
			every_scope_state = {
				limit = {
					has_modifier = nl_encouraged_proselytization
				}
				remove_modifier = nl_encouraged_proselytization
				add_modifier = {
					name = nl_encouraged_education
				}
			}
		}
		else_if = {
			limit = {
				has_variable = nl_gerudia_education_focus
			}
			every_scope_state = {
				limit = {
					has_modifier = nl_encouraged_education
				}
				remove_modifier = nl_encouraged_education
				add_modifier = {
					name = nl_encouraged_proselytization
				}
			}
		}
		if = {
			limit = {
				has_modifier = modifier_nl_conversion_costs
			}
			remove_modifier = modifier_nl_conversion_costs
		}
		ig:ig_devout = {
			add_modifier = {
				name = nl_ended_ravelian_mission_ig
				days = long_modifier_time
				is_decaying = yes
			}
		}
	}
}
