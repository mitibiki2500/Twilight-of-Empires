C14_reclaim_west = {
	name = "C14_reclaim_west_name"
	desc = "C14_reclaim_west_desc"

	visible = { always = yes }
	
	#Country scope
	ai_chance = {
		value = 0
		if = {
			limit = { infamy < 20 }
			add = 10
		}
	}

	possible = {
		hidden_trigger = { not = { has_variable = C14_reclaim_west_var } }
		any_scope_state = {
			state_region = s:STATE_CENTRAL_EFFELAI
			is_incorporated = yes
			is_split_state = no
		}
		any_scope_state = {
			state_region = s:STATE_NOGRUD
			is_incorporated = yes
			is_split_state = no
		}
		any_scope_state = {
			state_region = s:STATE_MARUKHAN
			is_incorporated = yes
			is_split_state = no
		}
	}

	effect = {
		set_variable = C14_reclaim_west_var
		change_infamy = 15
		s:STATE_SORFEN = {
			if = {
				limit = { any_scope_state = { NOT = { owner = ROOT } } }
				add_claim = ROOT 
			}
			add_homeland = cu:seedthrall
		}
		s:STATE_SEINAINE = { 
			if = {
				limit = { any_scope_state = { NOT = { owner = ROOT } } }
				add_claim = ROOT 
			}
			add_homeland = cu:seedthrall
		}
		s:STATE_DALAINE = { 
			if = {
				limit = { any_scope_state = { NOT = { owner = ROOT } } }
				add_claim = ROOT 
			}
			add_homeland = cu:seedthrall
		}
		s:STATE_THALASARAN = { 
			if = {
				limit = { any_scope_state = { NOT = { owner = ROOT } } }
				add_claim = ROOT 
			}
			add_homeland = cu:seedthrall
		}
		s:STATE_KIOHALEN = { 
			if = {
				limit = { any_scope_state = { NOT = { owner = ROOT } } }
				add_claim = ROOT 
			}
			add_homeland = cu:seedthrall
		}
		s:STATE_KIINDTIR = { 
			if = {
				limit = { any_scope_state = { NOT = { owner = ROOT } } }
				add_claim = ROOT 
			}
			add_homeland = cu:seedthrall
		}
		annex = c:C12
		annex = c:C13
		s:STATE_SCREAMING_JUNGLE = { 
			if = {
				limit = { 
					any_scope_state = { 
						NOT = {
							owner = ROOT
							owner = c:C12
							owner = c:C13
						}
					}
				}
				add_claim = ROOT 
			}
			add_homeland = cu:seedthrall
		}
		annex = c:C15
		s:STATE_WESTERN_EFFELAI = { 
			if = {
				limit = { 
					any_scope_state = { 
						NOT = {
							owner = ROOT
							owner = c:C15
						}
					}
				}
				add_claim = ROOT 
			}
		}
		annex = c:C17
		s:STATE_DEEPSONG = { 
			if = {
				limit = { 
					any_scope_state = { 
						NOT = {
							owner = ROOT
							owner = c:C17
						}
					}
				}
