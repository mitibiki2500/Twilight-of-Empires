#Clear the plants 1
scripted_button_varionail_plant_button_1 = {
    name = "scripted_button_varionail_plant_button_name"
	desc = "scripted_button_varionail_plant_button_desc"

    visible = {
        NOT = {
            has_variable = varionail_countermeasure_found_var
        }
    }

    possible = {
        custom_tooltip = {
            text = varionail_plant_cooldown_desc
            NOT = {
                has_variable = varionail_plant_cooldown
            }
        }
        country_has_building_group_levels = {
            type = bg_relics_digsite
            value >= 3
        }
    }

    effect = {
        set_variable = { name = varionail_plant_cooldown years = 2 }
        trigger_event = anb_varionail.003
    }

    ai_chance = {
        value = 20
    }
}
