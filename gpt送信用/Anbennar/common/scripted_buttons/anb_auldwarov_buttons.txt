sway_government_button = {
	name = "sway_government_button"
	desc = "sway_government_button_desc"

	visible = {
		always = yes
	}

	possible = {
		custom_tooltip = {
			text = ig_has_not_presented_demands_tt
			OR = {
				AND = {
					NOT = {
						has_variable = landowners_demands_presented
					}
					ig:ig_landowners ?= { is_in_government = yes }
				}
				AND = {
					NOT = {
						has_variable = armed_forces_demands_presented
					}
					ig:ig_armed_forces ?= { is_in_government = yes }
				}
				AND = {
					NOT = {
						has_variable = devout_demands_presented
					}
					ig:ig_devout ?= { is_in_government = yes }
				}
				AND = {
					NOT = {
						has_variable = rural_folk_demands_presented
					}
					ig:ig_rural_folk ?= { is_in_government = yes }
				}
				AND = {
					NOT = {
						has_variable = intelligentsia_demands_presented
					}
					ig:ig_intelligentsia ?= { is_in_government = yes }
				}
				AND = {
					NOT = {
						has_variable = industrialists_demands_presented
					}
					ig:ig_industrialists ?= { is_in_government = yes }
				}
				AND = {
					NOT = {
						has_variable = trade_unions_demands_presented
					}
					ig:ig_trade_unions ?= { is_in_government = yes }
				}
			}
		}
		NOT = { has_modifier = auldwarov_government_deliberations }
		government_legitimacy >= 50
	}

	effect = {
		custom_tooltip = {
			text = auldwarov_approach_current_government_tt
			if = {
				limit = {
					ig:ig_landowners ?= { is_in_government = yes }
				}
				set_variable = landowners_demands_presented
			}
			if = {
				limit = {
					ig:ig_intelligentsia ?= { is_in_government = yes }
				}
				set_variable = intelligentsia_demands_presented
			}
			if = {
				limit = {
					ig:ig_trade_unions ?= { is_in_government = yes }
				}
				set_variable = trade_unions_demands_presented
			}
			if = {
				limit = {
					ig:ig_industrialists ?= { is_in_government = yes }
				}
				set_variable = industrialists_demands_presented
			}
			if = {
				limit = {
					ig:ig_rural_folk ?= { is_in_government = yes }
				}
				set_variable = rural_folk_demands_presented
			}
			if = {
				limit = {
					ig:ig_devout ?= { is_in_government = yes }
				}
				set_variable = devout_demands_presented
			}
			if = {
				limit = {
					ig:ig_armed_forces ?= { is_in_government = yes }
				}
				set_variable = armed_forces_demands_presented
			}
		}
		trigger_event = {
            id = auldwarov.100
        }
	}

	ai_chance = {
	}
}

landowners_demands_button = {
	name = "landowners_demands_button"
	desc = "landowners_demands_button_desc"

	visible = {
		has_variable = landowners_demands_presented
		NOT = { has_variable = landowners_reneged }
	}

	possible = {
		custom_tooltip = {
			text = auldwarov_landowners_support_value_tt
			auldwarov_landowners_support >= 2
		}
		NOT = {
			has_modifier = auldwarov_landowners_swayed
		}
	}

	effect = {
		custom_tooltip = {
			text = set_landowners_swayed_tt
			set_variable = landowners_swayed_var
		}
		add_modifier = {
			name = auldwarov_landowners_swayed
		}
	}

	ai_chance = {
	}
}
