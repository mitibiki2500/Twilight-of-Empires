# Neratica-Minaria success
neratica_minaria_events.1 = {
	type = country_event
	placement = ROOT
	title = neratica_minaria_events.1.t
	desc = neratica_minaria_events.1.d
	flavor =  neratica_minaria_events.1.f

	duration = 3

	event_image = {
		video = "rp1_brazilian_city"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	trigger = {
		
	}

	immediate = {
		if = {
			limit = {
				any_subject_or_below = {
					c:B09 ?= this
					any_scope_state = {
						state_region = s:STATE_TAMETER
					}
				}
			}
			random_subject_or_below = {
				limit = {
					c:B09 ?= this
					any_scope_state = {
						state_region = s:STATE_TAMETER
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_TAMETER
					}
					save_scope_as = tameter_state
				}
			}
		}
		else_if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_TAMETER
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_TAMETER
				}
				save_scope_as = tameter_state
			}
		}
		else = {
			capital = {
				save_scope_as = tameter_state
			}
		}
	}

	option = { # Annex Minaria
		name = neratica_minaria_events.1.a
		trigger = {
			c:B27 ?= this
		}
		ai_chance = {
			base = 1
		}
		default_option = yes
		if = {
			limit = {
				c:B09 ?= {
					is_player = no
					is_subject_of = c:B27
				}
			}
			annex_with_incorporation = c:B09
		}
		change_tag = B69
		add_primary_culture = cu:minarian
		root = {
			save_scope_as = neratica_minaria_scope
		}
		every_country = {
			limit = {
				NOT = {
					THIS = ROOT
				}
				OR = {
					has_interest_marker_in_region = sr:region_noruin
					has_interest_marker_in_region = sr:region_haraf
				}
			}
			post_notification = neratica_minaria_success
		}
	}

	option = { #Note that it is possible to become Minaria again if you move capital and release Neratica, thus triggering the breakup event
		name = neratica_minaria_events.1.b
		trigger = {
			c:B09 ?= this
		}
		ai_chance = {
			base = 1
		}
		default_option = yes
		play_as = c:B27
		c:B27 ?= {
			if = {
				limit = {
					c:B09 ?= {
						is_subject_of = c:B27
					}
				}
				annex_with_incorporation = c:B09
				change_tag = B69
				add_primary_culture = cu:minarian
			}
		}
		root = {
			save_scope_as = neratica_minaria_scope
		}
		every_country = {
			limit = {
				NOT = {
					THIS = ROOT
				}
				OR = {
					has_interest_marker_in_region = sr:region_noruin
					has_interest_marker_in_region = sr:region_haraf
				}
			}
			post_notification = neratica_minaria_success
		}
	}
}

			is_subject_of = ROOT

		}

			leader = {

			add_amendment = {
				type = amendment_sovereign_of_the_nation

			}
		}
		ruler = {
			add_modifier = {
				name = modifier_emperor_of_the_nation
				days = long_modifier_time
			}
		}
		ig:ig_devout ?= {

				name = modifier_nerat_primacy_questioned
				days = long_modifier_time

				value = small_radicals
			}
		}
		every_country_in_south_na = {
			limit = {
				has_diplomatic_relevance = root
				NOR = {
					c:B09 ?= this
					c:B27 ?= this
				}
				NOT = {
					is_country_type = decentralized
				}
				is_subject = no	#Lorentish colonies shouldn't be involved in this
			}
			trigger_event = { id = neratica_minaria_events.5 days = 7 popup = yes }
		}
		ai_chance = {
			base = 25
		}
	}

	option = { #King of the Bone Citadel
		name = neratica_minaria_events.2.b
		ruler = {
			add_modifier = {
				name = modifier_new_onyx_king
				days = long_modifier_time
			}

		ig:ig_devout ?= {

			}

				add_modifier = {

#Neratican Republic
neratica_minaria_events.9 = {
	type = country_event
	placement = ROOT
	title = neratica_minaria_events.9.t
	desc = neratica_minaria_events.9.d
	flavor =  neratica_minaria_events.9.f

	duration = 3

	event_image = {
		video = "rp1_brazilian_city"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	trigger = {
		
	}

	immediate = {
		
	}

	option = {	#Time for a republic
		name = neratica_minaria_events.9.a
		default_option = yes
		ruler = {
			kill_character = yes
		}
		activate_law = law_type:law_presidential_republic
		if = {
			limit = {
				country_has_voting_franchise = yes
			}
			call_election = {
				months = 2
			}
		}
		add_modifier = {
                name = modifier_new_onyx_monarchy_broken
                days = long_modifier_time
            }
		add_radicals = {
			value = small_radicals
		}
		ig:ig_intelligentsia = {
			add_ruling_interest_group = yes
		}
		every_interest_group = {
			limit = {
				leader = {
					has_monarchist_ideology = yes
				}
			}
			leader = {
				set_ideology = ideology:ideology_theocrat
			}
		}
	}

	option = {	#Go back to the old theocracy
		name = neratica_minaria_events.9.b
		trigger = {
			ig:ig_devout ?= {
				is_marginal = no
			}
		}
		ruler = {
			kill_character = yes
		}
		activate_law = law_type:law_theocracy
		add_modifier = {
                name = modifier_new_onyx_monarchy_broken
                days = long_modifier_time
            }
		add_radicals = {
			value = medium_radicals
		}
		ig:ig_devout = {
			add_ruling_interest_group = yes
		}
		if = {
			limit = {
				NOT = {
					any_political_movement = {
						is_political_movement_type = movement_nihilist
					}
				}
			}
			create_political_movement = { type = movement_nihilist }
		}
		every_interest_group = {
			limit = {
				leader = {
					has_monarchist_ideology = yes
				}
			}
			leader = {
				set_ideology = ideology:ideology_moderate
			}
		}
	}
}
