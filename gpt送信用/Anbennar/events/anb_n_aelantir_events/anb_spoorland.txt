namespace = spoorland

# For Spoorland
spoorland.1 = {
	type = country_event
	placement = ROOT
	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	title = spoorland.1.t
	desc = spoorland.1.d
	flavor = spoorland.1.f

	duration = 3

	trigger = {
	}

	immediate = {
		c:B98 ?= {
			save_scope_as = concord_scope
		}
		c:A03 ?= {
			save_scope_as = lorent_scope
		}
	}

	option = {
		name = spoorland.1.a
		trigger = { is_player = no }
		c:B98 ?= {
			set_variable = spoorland_annex
			trigger_event = { id = spoorland.2 }
			show_as_tooltip = {
				c:B98 ?= {
					annex = ROOT
				}
			}
		}
		ai_chance = {
			base = 10
		}
	}
	option = {
		name = spoorland.1.b
		default_option = yes
		show_as_tooltip = {
			change_relations = {
				country = c:B98
				value = -30
			}
			s:STATE_SPOORLAND = {
				add_claim = c:B98
			}
		}
		c:B98 ?= {
			set_variable = spoorland_independent
			trigger_event = { id = spoorland.2 }
		}
		s:STATE_OKHIBOLI = {
			add_claim = c:B94
		}
		s:STATE_DIHEN = {
			add_claim = c:B94
		}
		s:STATE_GRAVEROADS = {
			add_claim = c:B94
		}
		add_modifier = {
			name = old_trail_state
			days = normal_modifier_time
		}
		ai_chance = {
			base = 0
		}
	}
}
