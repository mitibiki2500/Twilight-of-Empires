#Voices in government oppose collaboration with [rival]
anb_uneasy_peace.005 = {
    type = country_event
    placement = root

    title = anb_uneasy_peace.005.t
    desc = anb_uneasy_peace.005.d
    flavor = anb_uneasy_peace.005.f

    event_image = {
        video = "votp_agitator_speaking"
    }

    icon = "gfx/interface/icons/event_icons/event_protest.dds" 
    on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear" 

    duration = 3
    
    trigger = {
        #An IG exists with negative approval
        any_interest_group = {
            ig_approval < 0
            is_marginal = no
            is_member_of_any_lobby = no
        }
        OR = {
            c:B34 ?= this
            c:B41 ?= this
        }
    }

    cooldown = {
        years = 2
    }

    immediate = { #Save scope of the interest group to add to lobby
        random_interest_group = {
            limit = {
                ig_approval < 0
                is_marginal = no
                is_member_of_any_lobby = no
            }
            save_scope_as = expanse_opposing_ig
        }
        if = {
            limit = {
                c:B34 = this
            }
            c:B41 = {
                save_scope_as = expanse_protest_rival
            }
            c:B34 = {
                save_scope_as = expanse_protest_sender
            }
        }
        else = {
            c:B34 = {
                save_scope_as = expanse_protest_rival
            }
            c:B41 = {
                save_scope_as = expanse_protest_sender
            }
        }
        random_political_lobby = {
            limit = {
                is_political_lobby_type = lobby_anti_country
                target = { 
                    c:B34 = this
                }
            }
            save_scope_as = expanse_anti_lobby	
        }
    }

    option = { #Agree with the IG
        name = anb_uneasy_peace.005.a
        default_option = yes

        #Decrease relations with rival
        change_relations = {
            country = scope:expanse_protest_rival
            value = -10
        }
        #Gain loyalists in IG
        add_loyalists = {
            value = small_radicals
            interest_group = scope:expanse_opposing_ig
        }
        #Gain IG approval
        scope:expanse_opposing_ig = {
            add_modifier = {
                name = expanse_ig_supported
                years = 2
                is_decaying = no
            }
        }
        
        #Lose unity
        custom_tooltip = {
            text = uneasy_peace_unity_loss
            c:B41 = {
                je:je_uneasy_peace = { 
                    add_progress = { value = -5 name = uneasy_peace_bar }
                }
            }
            c:B34 = {
                je:je_uneasy_peace = { 
                    add_progress = { value = -5 name = uneasy_peace_bar }
                }
            }
            c:B42 = {
                je:je_uneasy_peace = { 
                    add_progress = { value = -5 name = uneasy_peace_bar }
                }
            }
        }
        #IG joins an anti-country lobby
        if = {
            limit = {
                any_political_lobby = {
					is_political_lobby_type = lobby_anti_country
					target = { 
						c:B34 = this
					}
				}
            }
            scope:expanse_anti_lobby = {
                add_lobby_member = scope:expanse_opposing_ig
            }
        }
        else = {
            create_political_lobby = {  
                type = lobby_anti_country  
                target = scope:expanse_protest_rival
                add_interest_group = scope:expanse_opposing_ig
                lobby_formation_reason = ideology
            }
        }
    }

    option = { #Ignore them
        name = anb_uneasy_peace.005.b
        #IG Joins an anti-country lobby
        if = {
            limit = {
                any_political_lobby = {
					is_political_lobby_type = lobby_anti_country
					target = { 
						c:B34 = this
					}
				}
            }
            scope:expanse_anti_lobby = {
                add_lobby_member = scope:expanse_opposing_ig
            }
        }
        else = {
            create_political_lobby = {  
                type = lobby_anti_country  
                target = scope:expanse_protest_rival
                add_interest_group = scope:expanse_opposing_ig
                lobby_formation_reason = ideology
            }
        }
        #Gain some radicals
        add_radicals = {
            value = small_radicals
            interest_group = scope:expanse_opposing_ig
        }
        #Lose some IG approval and weaken their political strength
        scope:expanse_opposing_ig = {
            add_modifier = {
                name = expanse_ig_ignored
                years = 2
                is_decaying = no
            }
        }
    }
    
    after = {
        #Give other tags a post notification
        c:B42 = {
            post_notification = expanse_anti_lobby_notification
        }
        scope:expanse_protest_rival = {
            post_notification = expanse_anti_lobby_notification
        }
    }
}
