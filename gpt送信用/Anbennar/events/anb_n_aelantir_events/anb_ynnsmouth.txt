			}

		}

		name = ynnsmouth_crisis.1.a

		}

		name = ynnsmouth_crisis.1.b

		first_valid = {
			triggered_desc = {
				desc = ynnsmouth_crisis.8.d1
				trigger = {
					has_variable = ynnsmouth_crisis_resolution_force
				}
			}
			triggered_desc = {
				desc = ynnsmouth_crisis.8.d2
				trigger = {
					has_variable = ynnsmouth_crisis_resolution_peaceful
				}

	trigger = {
	}

	immediate = {
		if = {
			limit = {
				any_scope_treaty = {
					binds = root

						has_type = transit_rights

					}
				}

					binds = root

					save_scope_as = transit_country_scope
				}
			}
		}
		else = {
			set_variable = ynnsmouth_crisis_resolution_force
		}
		c:B29 = {
			save_scope_as = sarda_scope
		}
	}

					limit = {
						owner = { owns_entire_state_region = STATE_YNNSMOUTH }
					}
				}

					value = 0.1
				}
				create_building={
					building="building_port"
					add_ownership={
						country={
							country="c:B29"
							levels=1
						}
					}
					reserves=1
				}
				create_building={
					building="building_trade_center"
					add_ownership={
						building={
							type="building_financial_district"
							country="c:B29"
							levels=1
							region="STATE_YNNSMOUTH"
						}
					} 
					reserves=1
				}
			}
		}
	}

	option = {
		name = ynnsmouth_crisis.8.b
		trigger = {
			owns_entire_state_region = STATE_YNNSMOUTH
		}
		add_modifier = {
			name = ynnsmouth_crisis_victory
			days = long_modifier_time
			is_decaying = yes
		}
		s:STATE_YNNSMOUTH = {
			random_scope_state = {
				if = {
					limit = {
						owner = { owns_entire_state_region = STATE_YNNSMOUTH }
					}
				}
				add_loyalists_in_state = {
					value = 0.1
				}
				create_building={
					building="building_port"
					add_ownership={
						country={
							country="c:B29"
							levels=1
						}
					}
					reserves=1
				}
				create_building={
					building="building_trade_center"
					add_ownership={
						building={
							type="building_financial_district"
							country="c:B29"
							levels=1
							region="STATE_YNNSMOUTH"
						}
					} 
					reserves=1
				}
			}

				state_region = s:STATE_YNNSMOUTH

		}

			remove_claim = root
		}

			type = catalyst_event_positive
			target = scope:transit_country_scope

			set_mutual_secret_goal = {

	option = {
		name = ynnsmouth_crisis.10.a
		s:STATE_YNNSMOUTH = {
			random_scope_state = {
				limit = {
					owner = ROOT
				}
				add_modifier = {
					name = ynnsmouth_boek_integration
					days = very_long_modifier_time
				}
			}
		}
		add_radicals = {
			value = small_radicals
			culture = cu:concordian
		}
	}

	option = {
		name = ynnsmouth_crisis.10.b

		if = {
			limit = {
				NOT = {
					any_political_movement = {

				}
			}
			create_political_movement = { type = movement_minority_rights }
		}
		else = {
			create_character = {
				culture = cu:boek
				is_agitator = yes
				interest_group = ig:ig_rural_folk
				ideology = ideology_reformer
				traits = {
					honorable
					experienced_political_operator
				}
				# on_created = { # unused
				# 	set_variable = ynnsmouthian_boek_agitator
				# }
			}
		}
	}
