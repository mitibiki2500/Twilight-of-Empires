#Starting Event
anb_varionail.001 = {
    type = country_event
    placement = root

    title = anb_varionail.001.t
    desc = anb_varionail.001.d
    flavor = anb_varionail.001.f

    event_image = {
        video = "unspecific_steam_ship"
    }

    icon = "gfx/interface/icons/event_icons/waving_flag.dds"
    on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear" 

    duration = 3
    
    immediate = {
        set_variable = {
            name = varionail_je_start_var
            value = yes
        }
    }

    option = { 
        name = anb_varionail.001.a
        default_option = yes
#        add_journal_entry = { type = je_varionail } TODO This wasn't working for some reason
        #Change primary culture
        add_primary_culture = cu:crescent_elf
        remove_primary_culture = cu:havener_elf
        #Do some initial assimilation
        custom_tooltip = {
            text = varionail_inital_assimilation_desc
            every_scope_state = {
                every_scope_pop = {
                    limit = {
                        this.culture = {
                            NOT = {
                                is_primary_culture_of = ROOT
                            }
                            shares_heritage_trait_with_any_primary_culture = ROOT
                        }
                    }
                    change_pop_culture = {
                        target = cu:crescent_elf
                        value = 0.5
                    }
                }
            }
        }
        s:STATE_VARIONAIL = {
            add_homeland = cu:crescent_elf
        }
        s:STATE_FALILTCOST = {
            add_homeland = cu:crescent_elf
        }
        s:STATE_SANCTUARY_FIELDS = {
            add_homeland = cu:crescent_elf
        }
        s:STATE_SILVERTENNAR = {
            add_homeland = cu:crescent_elf
        }

    }
}

anb_varionail.005 = {
    type = country_event
    placement = root

    title = anb_varionail.005.t
    desc = anb_varionail.005.d
    flavor = anb_varionail.005.f

    event_image = {
        video = "unspecific_fire"
    }

    icon = "gfx/interface/icons/event_icons/event_skull.dds"
    on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear" 

    duration = 3

    option = { #Varionail
        name = anb_varionail.005.a
        default_option = yes
        trigger = {
            s:STATE_VARIONAIL.region_state:B13 = {
                has_state_trait = state_trait_ravenous_isle
            }
        }
        s:STATE_VARIONAIL.region_state:B13 = {
            save_scope_as = varionail_clear_state
            add_devastation = 15
            add_modifier = {
                name = varionail_clearing
                months = 12
            }
        }
    }

    option = { #Silvertennar
        name = anb_varionail.005.b
        trigger = {
            s:STATE_SILVERTENNAR.region_state:B13 = {
                has_state_trait = state_trait_ravenous_isle
            }
        }
        s:STATE_SILVERTENNAR.region_state:B13 = {
            save_scope_as = varionail_clear_state
            add_devastation = 15
            add_modifier = {
                name = varionail_clearing
                months = 12
            }
        }
    }

    option = { #Faliltcost
        name = anb_varionail.005.c
        trigger = {
            s:STATE_FALILTCOST.region_state:B13 = {
                has_state_trait = state_trait_ravenous_isle
            }
        }
        s:STATE_FALILTCOST.region_state:B13 = {
            save_scope_as = varionail_clear_state
            add_devastation = 15
            add_modifier = {
                name = varionail_clearing
                months = 12
            }
        }
    }
    
    after = {
        trigger_event = { id = anb_varionail.006 months = 12 }
    }
}
#Clearing Plants (Complete)
anb_varionail.006 = {
    type = country_event
    placement = root

    title = anb_varionail.006.t
    desc = anb_varionail.006.d
    flavor = anb_varionail.006.f

    event_image = {
        video = "unspecific_fire"
    }

    icon = "gfx/interface/icons/event_icons/event_industry.dds"
    on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear" 

    duration = 3

    option = {
        name = anb_varionail.006.a
        default_option = yes
        scope:varionail_clear_state = {
            remove_state_trait = state_trait_ravenous_isle
            if = {
                limit = {
                    c:B13 ?= {
                        has_variable = varionail_countermeasure_clear_var
                    }
                }
                add_state_trait = state_trait_ravenous_isle_cleared
            }
            else = {
                add_state_trait = state_trait_ravenous_isle_controlled
            }
        }
    }
}

anb_varionail.009 = {
    type = country_event
    placement = scope:varionail_random_event_state

    title = anb_varionail.009.t
    desc = anb_varionail.009.d
    flavor = anb_varionail.009.f

    event_image = {
        video = "africa_construction_colony"
    }

    icon = "gfx/interface/icons/event_icons/event_skull.dds" 
    on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear" 

    duration = 3

    trigger = {
        NOT = {
            has_variable = varionail_009_cooldown
        }
        any_scope_state = {
            has_state_trait = state_trait_ravenous_isle
        }
    }

    immediate = {
        c:B13 ?= {
            random_scope_state = {
                limit = {
                    has_state_trait = state_trait_ravenous_isle
                }
                save_scope_as = varionail_random_event_state
            }
        }
        set_variable = { name = varionail_009_cooldown years = 1 }
    }

    option = { #Favor Elves
        name = anb_varionail.009.a
        default_option = yes
        scope:varionail_random_event_state = {
            #Human pops gain radicals
            custom_tooltip = {
                text = varionail_human_radicals_desc
                add_radicals_in_state = {
                    value = small_radicals
                    culture = cu:pearlsedger
                }
                add_radicals_in_state = {
                    value = small_radicals
                    culture = cu:vernman
                }
                add_radicals_in_state = {
                    value = small_radicals
                    culture = cu:west_damerian
                }
                add_radicals_in_state = {
                    value = small_radicals
                    culture = cu:lorentish
                }
            }
            #Elven pops get +1 SOL
            add_modifier = {
                name = varionail_elven_sol_boost
                months = 24
            }
            #Human pops get -1 SOL
            add_modifier = {
                name = varionail_human_sol_decrease
                months = 24
            }
        }
    }

    option = { #Favor Humans
        name = anb_varionail.009.b
        scope:varionail_random_event_state = {
            add_radicals_in_state = {
                value = small_radicals
                #Only for elves
                culture = cu:crescent_elf
            }
            #Elven pops get -1 SOL
            add_modifier = {
                name = varionail_elven_sol_decrease
                months = 24
            }
            #Human pops get +1 SOL
            add_modifier = {
                name = varionail_human_sol_boost
                months = 24
            }
        }
    }
}
