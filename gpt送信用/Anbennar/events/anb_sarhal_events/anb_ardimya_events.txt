#[Admiral] Pushes for Naval Advancement
anb_ardimya.006 = {
    type = country_event
    placement = root

    title = anb_ardimya.006.t  
    desc = anb_ardimya.006.d  
    flavor = anb_ardimya.006.f  

    event_image = {
        video = "unspecific_gears_pistons" 
    }

    icon = "gfx/interface/icons/event_icons/event_military.dds"

    on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    duration = 3

    trigger = {
        NOT = {
            has_variable = ardimya_tech_event_cooldown_var
        }
    }

    immediate = {
        set_variable = {
            name = ardimya_tech_event_cooldown_var
            years = 3
        }
    }

    option = { #Invest
        name = anb_ardimya.006.a
        default_option = yes
        #Add weekly expenses for 2 years and Armed Forces Gain Approval
        add_modifier = {
            name = ardimya_navy_investment
            years = 2
        }
        #Admiral gains popularity
        random_scope_admiral = {
            add_modifier = {
                name = ardimya_navy_invest_advice_listened
                years = 2
            }
        }
        #Add some tech progress
        if = {
            limit = {
                NOT = {
                    has_technology_researched = hydraulic_cranes
                }
            }
            add_technology_progress = { progress = 500 technology = hydraulic_cranes }
        }
        else_if = {
            limit = {
                NOT = {
                    has_technology_researched = screw_frigate
                }
            }
            add_technology_progress = { progress = 500 technology = screw_frigate }
        }
        else_if = {
            limit = {
                NOT = {
                    has_technology_researched = power_of_the_purse
                }
            }
            add_technology_progress = { progress = 500 technology = power_of_the_purse }
        }
        else_if = {
            limit = {
                NOT = {
                    has_technology_researched = ironclad_tech
                }
            }
            add_technology_progress = { progress = 500 technology = ironclad_tech }
        }
        else_if = {
            limit = {
                NOT = {
                    has_technology_researched = gantry_cranes
                }
            }
            add_technology_progress = { progress = 500 technology = gantry_cranes }
        }
        else = {
            add_technology_progress = { progress = 500 technology = jeune_ecole }
        }
    }

    option = { #Do not Invest
        name = anb_ardimya.006.b
        default_option = yes
        #Armed Forces Lose Approval
        add_modifier = {
            name = ardimya_navy_no_investment
            years = 2
        }
        #Admiral Loses Popularity
        random_scope_admiral = {
            add_modifier = {
                name = ardimya_navy_invest_advice_ignored
                years = 2
            }
        }
    }
}
