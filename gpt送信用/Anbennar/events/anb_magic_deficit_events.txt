magic_deficit.1 = { #Shortage of Transmuters - Production level 1
	type = country_event
	placement = scope:shortage_state

	title = magic_deficit.1.t
	desc = magic_deficit.1.d
	flavor = magic_deficit.1.f

	event_image = {
		video = "middleeast_engineer_blueprint"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	trigger = {
		has_modifier = country_magical_expertise_deficit_malus_modifier
		has_variable = production_magic_spell
		any_scope_state = {
			has_building = building_urban_center
			NOT = {has_modifier = modifier_state_transmuter_shortage}
		}
	}
	
	immediate = {
		random_scope_state = {
			limit = {
				has_building = building_urban_center
				NOT = {has_modifier = modifier_state_transmuter_shortage}
			}
			save_scope_as = shortage_state
		}
	}
	
	option = {
		name = magic_deficit.1.a
		default_option = yes
		scope:shortage_state = {
			add_modifier = {
				name = modifier_state_transmuter_shortage
				years = 10
				is_decaying = yes 
				multiplier = 2
			}
		}
	}
	option = {
		name = magic_deficit.1.b
		custom_tooltip = {
			text = drop_magic_production_level_tt
			drop_production_spell_level = yes
		}
		add_modifier = {
			name = modifier_magical_bureaucracy
			multiplier = 5
			years = 5
		}
	}
	option = {
		name = magic_deficit.1.c
		trigger = {has_law_or_variant = law_type:law_interventionism}
		add_modifier = {
			name = modifier_transmuter_subsidies
			years = 5
			multiplier = money_amount_multiplier_small
		}
		scope:shortage_state = {
			add_modifier = {
				name = modifier_state_transmuter_shortage
				years = 5
				is_decaying = yes
			}
		}
	}
}

magic_deficit.7 = { #Magical Detonation - Production Level 3
	type = country_event
	placement = scope:explosion_state
	
	title = magic_deficit.7.t
	desc = magic_deficit.7.d
	flavor = magic_deficit.7.f
	
	event_image = {
		video = "unspecific_devastation"
	}
	
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	
	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	duration = 3
	
	trigger = {
		has_modifier = country_magical_expertise_deficit_malus_modifier
		has_variable = production_magic_spell
		country_normalized_magical_expertise_deficit = 1
		any_scope_state = { devastation <= 0.1 }
	}
	
	immediate = {
		random_scope_state = { #Can't just weight by infrastructure normally and I've spent way to long on this so I'm being evil instead
			limit = {devastation <= 0.1}
			weight = {
				base = 1
				modifier = { #10
					infrastructure_usage > 10
					add = 10
				}
				modifier = { #25
					infrastructure_usage > 25
					add = 50
				}
				modifier = { #100
					infrastructure_usage > 100
					add = 400
				}
				modifier = { #300
					infrastructure_usage > 300
					add = 1200
				}
				modifier = { #1000
					infrastructure_usage > 1000
					add = 5000
				}
				modifier = { #1500
					infrastructure_usage > 1500
					add = 9000
				}
				modifier = { #3000
					infrastructure_usage > 3000
					add = 21000
				}
			}
			save_scope_as = explosion_state
		}
	}
	
	option = {
		name = magic_deficit.7.a
		default_option = yes
		scope:explosion_state = {
			add_devastation = 70
			add_radicals = {value = 0.1}
		}
		add_modifier = {
			name = modifier_explosion_subsidies
			multiplier = money_amount_multiplier_medium
			years = 10
		}
	}
	
	option = {
		name = magic_deficit.7.b
		custom_tooltip = {
			text = drop_magic_production_level_tt
			drop_production_spell_level = yes
		}
		scope:explosion_state = {
			add_devastation = 35
			add_radicals = {value = 0.15}
		}
	}

	option = {
		name = magic_deficit.7.c
		trigger = {
			has_law_or_variant = law_type:law_national_guard
			institution_investment_level = {
				institution = institution_home_affairs
				value = 3
			}
		}
		add_modifier = {
			name = modifier_magical_bureaucracy
			years = 10
			multiplier = 8
			is_decaying = no
		}
		scope:explosion_state = {
			add_devastation = 35
			add_radicals = {value = 0.05}
		}
	}
}

		}
	}
	
	option = {
		name = magic_deficit.9.a

		}
	}

		}
		every_political_movement = {
			limit = {is_political_movement_type = movement_magocratic}
			add_modifier = {
				name = modifier_small_magocratic_rebellion
				years = 10
			}
		}
	}
    	
	option = {
		name = magic_deficit.9.c
