# JE start
broken_empire.1 = {
	type = country_event
	placement = root
	
	title = broken_empire.1.t
	desc =  broken_empire.1.d
	flavor = broken_empire.1.f
	
	event_image = {
		video = "europenorthamerica_russian_serfs"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	
	duration = 3
	
	trigger = {
		c:A01 ?= this
		has_technology_researched = nationalism
		NOT = { has_technology_researched = pan-nationalism }
		NOT = { has_variable = anbennar_var }
	}

	immediate = {
		root = { save_scope_as = broken_empire_target }
		set_variable = {
			name = anbennar_var
			value = 0
		}
		set_global_variable = {
			name = broken_empire_time_var
			years = 15
		}
		set_global_variable = {
			name = broken_empire_unrest_var
			years = 10
		}
		set_variable = {
			name = broken_empire_start_year
			value = year
		}
	}

	option = { 
		name = broken_empire.1.a
		default_option = yes
		add_journal_entry = { type = je_broken_empire }
		custom_tooltip = {
			text = broken_empire_increase_600_tt
			change_variable = { name = anbennar_var add = 600 }
		}
		every_scope_culture = {
			limit = {
				has_discrimination_trait = heritage_anbennarian
				NOT = { this = cu:costine }
			}
			ROOT = {
				add_radicals = {
					value = large_radicals
					culture = prev
				}
				create_political_movement = { type = movement_cultural_minority culture = prev }
			}
		}
	}	
}

								}
							}

								}

								}
							}
						}
					}

								is_subject_of = c:A01
								liberty_desire >= 50
							}

			}

			random_diplomatic_play = {
				limit = {
					is_diplomatic_play_type = dp_independence
					initiator = {
						scope:anbennar_indep_leader ?= this
					}
				}
				add_initiator_backers = { prev }
			}
		}
		set_variable = {
			name = anbennar_big_war
		}
	}
