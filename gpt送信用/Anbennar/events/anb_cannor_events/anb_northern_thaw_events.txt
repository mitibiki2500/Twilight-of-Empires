#League breakup
northern_thaw_events.2 = {
	type = country_event
	placement = root

	title = northern_thaw_events.2.t
	desc = northern_thaw_events.2.d
	flavor = northern_thaw_events.2.f

	duration = 3

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	event_image = {
		video = "europenorthamerica_springtime_of_nations"
	}

	trigger = {
		
	}

	option = {
		name = northern_thaw_events.2.a
		ai_chance = {
			base = 1
		}
		default_option = yes

		every_subject_or_below = {
			limit = {
				capital.region = sr:region_gerudia	#includes the Reach
			}
			make_independent = yes
		}

		#Gerudia secession
		if = {
			limit = {
				NOT = {
					country_has_primary_culture = cu:dalr
				}
			}
			if = {
				limit = { exists = s:STATE_BJARNLAND.region_state:A04 }
				s:STATE_BJARNLAND.region_state:A04 = {
					save_scope_as = gerudia_secession_state
				}
			}
			else = {
				random_scope_state = {
					limit = {
						state_region = {
							is_homeland = dalr
						}
					}
					save_scope_as = gerudia_secession_state
				}
			}
			create_country = {
				tag = A62
				origin = this
				state = scope:gerudia_secession_state
			}
			every_scope_state = {
				limit = {
					OR = {
						state_region = { is_homeland = cu:dalr }
						state_region = { is_homeland = cu:olavish }
					}
				}
				set_state_owner = c:A62
			}
		}

		#Reach secession - but keep straits if Gerudia is integrated (so that things look saner)
		if = {
			limit = {
				NOT = {
					country_has_primary_culture = cu:blue_reachman
				}
				country_has_primary_culture = cu:dalr
			}
			if = {
					limit = { exists = s:STATE_BAYVEK.region_state:A04 }
					s:STATE_BAYVEK.region_state:A04 = {
						save_scope_as = reach_secession_state
					}
				}
				else = {
					random_scope_state = {
						limit = {
							state_region = {
								is_homeland = blue_reachman
							}
							NOT = {
								state_region = s:STATE_SERPENTSHEAD
							}
						}
						save_scope_as = reach_secession_state
					}
				}
				create_country = {
					tag = A36
					origin = this
					state = scope:reach_secession_state
				}
				every_scope_state = {
					limit = {
						OR = {
							state_region = { is_homeland = cu:blue_reachman }
							state_region = { is_homeland = cu:white_reachman }
						}
						NOT = {
							state_region = s:STATE_SERPENTSHEAD
						}
					}
					set_state_owner = c:A36
				}
				#it's kept so that the country stays connected (the Dalr garrison helps ?) but they aren't happy about it at all
				if = {
					limit = { exists = s:STATE_SERPENTSHEAD.region_state:A04 }
					s:STATE_SERPENTSHEAD.region_state:A04 = {
						add_radicals_in_state = {
							value = very_large_radicals
						}
					}
				}
		}
		else_if = {
			#Reach secession
			if = {
				limit = {
					NOT = {
						country_has_primary_culture = cu:blue_reachman
						country_has_primary_culture = cu:dalr
					}
				}
				if = {
					limit = { exists = s:STATE_BAYVEK.region_state:A04 }
					s:STATE_BAYVEK.region_state:A04 = {
						save_scope_as = reach_secession_state
					}
				}
				else = {
					random_scope_state = {
						limit = {
							state_region = {
								is_homeland = blue_reachman
							}
						}
						save_scope_as = reach_secession_state
					}
				}
				create_country = {
					tag = A36
					origin = this
					state = scope:reach_secession_state
				}
				every_scope_state = {
					limit = {
						OR = {
							state_region = { is_homeland = cu:blue_reachman }
							state_region = { is_homeland = cu:white_reachman }
						}
					}
					set_state_owner = c:A36
				}
			}
		}

		#turns back into Gawed via dynamic naming

		add_radicals = {
			value = very_large_radicals
			culture = cu:gawedi
		}
	}
}

			limit = {
