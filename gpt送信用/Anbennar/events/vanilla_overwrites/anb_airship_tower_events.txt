# Airship Tower complete
airship_tower.2 = {
	type = country_event

	title = airship_tower.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = airship_tower.2.d
				trigger = {
					NOT = { has_variable = first_airship_tower_builder }
				}
			}
			triggered_desc = {
				desc = airship_tower.2.d2
				trigger = {
					has_variable = first_airship_tower_builder
				}
			}
		}
	}
	flavor = airship_tower.2.f

	placement = scope:airship_tower_state

	event_image = {
		video = "unspecific_world_fair"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	trigger = {
		# triggered by je_airship_tower_construction
	}

	immediate = {
		random_scope_state = {
			limit = {
				has_building = building_airship_tower
			}
			save_scope_as = airship_tower_state
			random_scope_building = {
				limit = {
					is_building_type = building_airship_tower
				}
				save_scope_as = airship_tower_building_scope
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = first_airship_tower }
			}
			set_global_variable = first_airship_tower
			set_variable = first_airship_tower_builder
		}
	}

	option = {
		name = airship_tower.2.a
        default_option = yes
        trigger = {
        	NOT = { has_variable = first_airship_tower_builder }
        }
        add_modifier = {
        	name = airship_tower_modifier
        	days = normal_modifier_time
        }
	}

	option = {
		name = airship_tower.2.b
        default_option = yes
        highlighted_option = yes
        trigger = {
        	custom_tooltip = {
        		text = first_airship_tower_tt
        		has_variable = first_airship_tower_builder
        	}
        }
        add_modifier = {
        	name = airship_tower_modifier
        	days = normal_modifier_time
        }
        add_modifier = { # prestige
        	name = first_airship_tower_modifier
        	days = normal_modifier_time
        }
	}
}
