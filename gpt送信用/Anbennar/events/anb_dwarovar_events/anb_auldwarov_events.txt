#Radicals consolidate into an unaccepted dwarf culture
auldwarov.5 = {
	type = country_event
	placement = ROOT

	title = auldwarov.5.t
	desc = auldwarov.5.d
	flavor = auldwarov.5.f
	
	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}
	
	duration = 3
	
	trigger = {
		has_journal_entry = je_aul_dwarov
		any_scope_state = {
			has_state_trait = state_trait_dwarven_hold
			turmoil >= 0.2
			NOT = { is_homeland_of_country_cultures = root }
			any_scope_pop = {
				this.culture = {
					NOT = {
						is_primary_culture_of = ROOT
					}
					has_discrimination_trait_group = heritage_group_dwarf
				}
			}
		}
	}
	
	immediate = {
		random_scope_state = {
			limit = {
				has_state_trait = state_trait_dwarven_hold
				turmoil >= 0.2
				NOT = { is_homeland_of_country_cultures = root }
				any_scope_pop = {
					this.culture = {
						NOT = {
							is_primary_culture_of = ROOT
						}
						has_discrimination_trait_group = heritage_group_dwarf
					}
				}
			}
			save_scope_as = event_hold_scope
			hold_culture_check = yes
		}
	}
	
	option = {
		default_option = yes
		
		name = auldwarov.5.a
		custom_tooltip = {
			text = hold_culture_shift_radical_tt
			scope:event_hold_scope = {
				every_scope_pop = {
					limit = {
						culture = { has_discrimination_trait_group = heritage_group_dwarf }
						total_size <= 20000
					}
					change_pop_culture = {
						target = scope:event_hold_culture_scope
						value = 1
					}
				}
				every_scope_pop = {
					limit = {
						culture = { has_discrimination_trait_group = heritage_group_dwarf }
					}
					change_pop_culture = {
						target = scope:event_hold_culture_scope
						value = 0.15
					}
				}
			}
		}
	}
}

auldwarov.7 = {
	type = country_event
	placement = ROOT

	title = auldwarov.7.t
	desc = auldwarov.7.d
	flavor = auldwarov.7.f
	
	event_image = {
		video = "ep1_printing_press"
	}
	
	duration = 3
	
	trigger = {
		has_journal_entry = je_aul_dwarov
		NOT = { has_variable = hold_migration_opposed }
		any_scope_state = {
			AND = {
				has_state_trait = state_trait_dwarven_hold
				NOT = { state_region = s:STATE_MARRHOLD }
			}
			NOT =	{
				any_scope_pop = {
					this.culture = {
						has_discrimination_trait_group = heritage_group_dwarf
					}
					total_size >= 20000
				}
			}
		}
	}
	
	immediate = {
		random_scope_state = {
			limit = {
				has_state_trait = state_trait_dwarven_hold
				NOT =	{
					any_scope_pop = {
						this.culture = {
							has_discrimination_trait_group = heritage_group_dwarf
						}
						total_size >= 20000
					}
				}
			}
			save_scope_as = event_hold_scope
			hold_culture_check = yes
		}
		random_scope_state = {
			limit = {
				is_homeland_of_country_cultures = ROOT
				any_scope_pop = {
					this.culture = {
						has_discrimination_trait_group = heritage_group_dwarf
					}
					total_size >= 200000
				}
			}
			save_scope_as = migration_origin
		}
		random_scope_state = {
			limit = {
				is_homeland_of_country_cultures = ROOT
				any_scope_pop = {
					this.culture = scope:event_hold_culture_scope
				}
			}
			save_scope_as = migration_origin
		}
		scope:migration_origin = {
			every_scope_pop = {
				limit = {
					culture = { has_discrimination_trait_group = heritage_group_dwarf }
					total_size >= 1000000
				}
				change_pop_culture = {
					target = scope:event_hold_culture_scope
					value = 0.005
				}
			}
			every_scope_pop = {
				limit = {
					culture = { has_discrimination_trait_group = heritage_group_dwarf }
					total_size >= 500000
				}
				change_pop_culture = {
					target = scope:event_hold_culture_scope
					value = 0.01
				}
			}
			every_scope_pop = {
				limit = {
					culture = { has_discrimination_trait_group = heritage_group_dwarf }
					total_size >= 200000
				}
				change_pop_culture = {
					target = scope:event_hold_culture_scope
					value = 0.02
				}
			}
		}
	}
	
	option = {
		default_option = yes
		
		name = auldwarov.7.a
		custom_tooltip = {
			text = hold_migrants_let_go_tt
			scope:migration_origin = {
				every_scope_pop = {
					limit = {
						culture = scope:event_hold_culture_scope
						total_size <= 30000
					}
					move_partial_pop = {
						state = scope:event_hold_scope
						population_ratio = 1
					}
				}
				every_scope_pop = {
					limit = {
						culture = scope:event_hold_culture_scope
						total_size <= 200000
					}
					move_partial_pop = {
						state = scope:event_hold_scope
						population_ratio = 0.3
					}
				}
				every_scope_pop = {
					limit = {
						culture = scope:event_hold_culture_scope
						total_size <= 1000000
					}
					move_partial_pop = {
						state = scope:event_hold_scope
						population_ratio = 0.05
					}
				}
			}
			scope:event_hold_scope = {
				add_loyalists_in_state = {
					value = 0.1
					culture = scope:event_hold_culture_scope
				}
			}
		}
		scope:event_hold_scope = {
			add_radicals_in_state = {
				value = 0.05
			}
		}
	}

	option = {
		name = auldwarov.7.b
		custom_tooltip = {
			text = hold_migrants_forced_stay_tt
			set_variable = {
				name = hold_migration_opposed
				months = 24
			}
		}
		scope:migration_origin = {
			add_radicals_in_state = {
				value = 0.2
				culture = scope:event_hold_culture_scope
			}
		}
	}
}
